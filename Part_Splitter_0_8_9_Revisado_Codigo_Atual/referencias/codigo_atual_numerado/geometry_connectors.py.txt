0025: def point_in_polygon_2d(x: float, y: float, poly: List[Tuple[float, float]]) -> bool:
0026:     """Teste de inclusão de ponto 2D em polígono via Ray Casting (Teorema de Jordan)."""
0027:     inside = False
0028:     n = len(poly)
0029:     if n < 3:
0030:         return False
0031:     j = n - 1
0032:     for i in range(n):
0033:         xi, yi = poly[i]
0034:         xj, yj = poly[j]
0035:         if ((yi > y) != (yj > y)) and (x < (xj - xi) * (y - yi) / (yj - yi + 1e-12) + xi):
0036:             inside = not inside
0037:         j = i
0038:     return inside

0041: def point_to_segment_dist_sq(px: float, py: float, x1: float, y1: float, x2: float, y2: float) -> float:
0042:     """Calcula o quadrado da distância euclidiana de um ponto 2D a um segmento de reta."""
0043:     dx = x2 - x1
0044:     dy = y2 - y1
0045:     l2 = dx * dx + dy * dy
0046:     if l2 < 1e-12:
0047:         return (px - x1) ** 2 + (py - y1) ** 2
0048:     t = max(0.0, min(1.0, ((px - x1) * dx + (py - y1) * dy) / l2))
0049:     proj_x = x1 + t * dx
0050:     proj_y = y1 + t * dy
0051:     return (px - proj_x) ** 2 + (py - proj_y) ** 2

0054: def min_distance_to_polygon_boundaries(
0055:     px: float,
0056:     py: float,
0057:     outer_polys: List[List[Tuple[float, float]]],
0058:     hole_polys: List[List[Tuple[float, float]]],
0059: ) -> float:
0060:     """Calcula a menor distância euclidiana de um ponto até as bordas da interface (B06 / D2)."""
0061:     min_d_sq = 1e18
0062:     for poly in outer_polys:
0063:         m = len(poly)
0064:         for i in range(m):
0065:             p1 = poly[i]
0066:             p2 = poly[(i + 1) % m]
0067:             d_sq = point_to_segment_dist_sq(px, py, p1[0], p1[1], p2[0], p2[1])
0068:             if d_sq < min_d_sq:
0069:                 min_d_sq = d_sq
0070:     for h_poly in hole_polys:
0071:         m = len(h_poly)
0072:         for i in range(m):
0073:             p1 = h_poly[i]
0074:             p2 = h_poly[(i + 1) % m]
0075:             d_sq = point_to_segment_dist_sq(px, py, p1[0], p1[1], p2[0], p2[1])
0076:             if d_sq < min_d_sq:
0077:                 min_d_sq = d_sq
0078:     return math.sqrt(min_d_sq) if min_d_sq < 1e17 else 0.0

0081: def find_optimal_grid_axes(

0084:     v_axis: Vector,
0085: ) -> Tuple[Vector, Vector, float, float, float, float]:

0087:

0090:     """
0091:     n_pts = len(outer_poly)
0092:     if n_pts < 3:

0095:     best_area = 1e18
0096:     best_angle = 0.0
0097:     best_min_u, best_max_u = 0.0, 0.0
0098:     best_min_v, best_max_v = 0.0, 0.0
0099: 
0100:     # Testa os ângulos de cada aresta do polígono
0101:     tested_angles = [0.0]
0102:     for i in range(n_pts):
0103:         p1 = outer_poly[i]
0104:         p2 = outer_poly[(i + 1) % n_pts]
0105:         dx = p2[0] - p1[0]
0106:         dy = p2[1] - p1[1]
0107:         length = math.hypot(dx, dy)
0108:         if length > 1e-6:
0109:             ang = math.atan2(dy, dx) % (math.pi / 2.0)
0110:             tested_angles.append(ang)
0111: 
0112:     for ang in tested_angles:
0113:         cos_a = math.cos(ang)
0114:         sin_a = math.sin(ang)
0115:         proj_u = [x * cos_a + y * sin_a for x, y in outer_poly]
0116:         proj_v = [-x * sin_a + y * cos_a for x, y in outer_poly]
0117:         w = max(proj_u) - min(proj_u)
0118:         h = max(proj_v) - min(proj_v)
0119:         area = w * h
0120:         if area < best_area - 1e-6:
0121:             best_area = area
0122:             best_angle = ang
0123:             best_min_u, best_max_u = min(proj_u), max(proj_u)
0124:             best_min_v, best_max_v = min(proj_v), max(proj_v)
0125: 
0126:     cos_b = math.cos(best_angle)
0127:     sin_b = math.sin(best_angle)
0128:     grid_u_axis = (u_axis * cos_b + v_axis * sin_b).normalized()
0129:     grid_v_axis = (-u_axis * sin_b + v_axis * cos_b).normalized()
0130: 
0131:     span_u = best_max_u - best_min_u
0132:     span_v = best_max_v - best_min_v
0133:     center_u = (best_min_u + best_max_u) * 0.5
0134:     center_v = (best_min_v + best_max_v) * 0.5
0135: 
0136:     return grid_u_axis, grid_v_axis, span_u, span_v, center_u, center_v

0139: def create_capsule_2d_profile(w: float, half_len: float, segments: int = 32, offset: float = 0.0) -> List[Tuple[float, float]]:
0140:     """Gera pontos 2D fechados e ordenados para o perfil de estádio/cápsula com suporte a offset uniforme.
0141:     
0142:     Inclui os pontos finais de tangência para que os lados retos sejam estritamente paralelos,
0143:     garantindo folga perimetral uniforme sem segmentos diagonais indevidos.
0144:     """
0145:     half_segs = max(8, segments // 2)
0146:     r = max(1e-5, w - offset)
0147:     if half_len < 1e-6:
0148:         # Círculo perfeito
0149:         return [(r * math.cos(2.0 * math.pi * i / segments), r * math.sin(2.0 * math.pi * i / segments)) for i in range(segments)]
0150: 
0151:     pts = []
0152:     # Arco superior: centro (0, half_len), ângulo de 0 a pi (incluindo extremos tangentes)
0153:     for i in range(half_segs + 1):
0154:         ang = math.pi * (i / half_segs)
0155:         pts.append((r * math.cos(ang), half_len + r * math.sin(ang)))
0156:     # Arco inferior: centro (0, -half_len), ângulo de pi a 2*pi (incluindo extremos tangentes)
0157:     for i in range(half_segs + 1):
0158:         ang = math.pi + math.pi * (i / half_segs)
0159:         pts.append((r * math.cos(ang), -half_len + r * math.sin(ang)))
0160:     return pts

0222:             cos_a = math.cos(ang)
0223:             sin_a = math.sin(ang)

0235:     for i in range(num_v):
0236:         i_next = (i + 1) % num_v

0729: def determine_connector_material(
0730:     obj: bpy.types.Object,
0731:     positions: List[Vector],
0732:     connector_radius_bu: float,
0733:     plane_normal: Vector,
0734: ) -> Optional[bpy.types.Material]:
0735:     """Determina o material a ser atribuído aos pinos conforme a regra C06.
0736:     
0737:     1. Verifica o material na área real de ligação dos pinos no modelo (world space).
0738:        Se todos os pontos de contato compartilham um único material, usa esse material.
0739:     2. Se houver múltiplos materiais no contato ou ambiguidade, calcula o material
0740:        predominante da peça por soma de área das faces em World Space.
0741:     3. Em caso de empate, usa o menor índice de material estável.
0742:     """
0743:     if not obj or not obj.data or not getattr(obj.data, "materials", None):
0744:         return None
0745: 
0746:     valid_mats = [m for m in obj.data.materials if m is not None]
0747:     if not valid_mats:
0748:         return None
0749:     if len(valid_mats) == 1:
0750:         return valid_mats[0]
0751: 
0752:     mw = obj.matrix_world
0753:     mesh = obj.data
0754: 
0755:     # 1. Amostragem local nos pontos de ligação dos pinos
0756:     local_mats = set()
0757:     r_sq = (connector_radius_bu * 1.5) ** 2
0758: 
0759:     for poly in mesh.polygons:
0760:         poly_center_world = mw @ poly.center
0761:         if any((poly_center_world - pos).length_squared <= r_sq for pos in positions):
0762:             mat_idx = poly.material_index
0763:             if 0 <= mat_idx < len(mesh.materials):
0764:                 m = mesh.materials[mat_idx]
0765:                 if m:
0766:                     local_mats.add(m)
0767: 
0768:     if len(local_mats) == 1:
0769:         return next(iter(local_mats))
0770: 
0771:     # 2. Fallback C06: Material predominante por área total das faces em World Space
0772:     mat_areas = {}
0773:     for poly in mesh.polygons:
0774:         mat_idx = poly.material_index
0775:         if 0 <= mat_idx < len(mesh.materials):
0776:             m = mesh.materials[mat_idx]
0777:             if m:
0778:                 verts_w = [mw @ mesh.vertices[vi].co for vi in poly.vertices]
0779:                 poly_area = 0.0
0780:                 v0 = verts_w[0]
0781:                 for i in range(1, len(verts_w) - 1):
0782:                     v1 = verts_w[i]
0783:                     v2 = verts_w[i + 1]
0784:                     poly_area += ((v1 - v0).cross(v2 - v0)).length * 0.5
0785:                 mat_areas[m] = mat_areas.get(m, 0.0) + poly_area
0786: 
0787:     if not mat_areas:
0788:         return valid_mats[0]
0789: 
0790:     sorted_mats = sorted(
0791:         mat_areas.keys(),
0792:         key=lambda m: (-mat_areas[m], list(mesh.materials).index(m))
0793:     )
0794:     return sorted_mats[0]
