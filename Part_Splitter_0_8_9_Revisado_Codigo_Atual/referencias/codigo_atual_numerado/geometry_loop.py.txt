0699:                 f"Recomenda-se selecionar um contorno na base plana."

0701:         else:
0702:             reason = (
0703:                 f"Saliência 3D detectada (altura saliente: {span_h_mm:.2f} mm). "
0704:                 f"Destacando volume pela interface do contorno."
0705:             )
0706:         is_planar_region = False
0707: 
0708:     return {
0709:         "detected_mode": detected_mode,
0710:         "reason": reason,
0711:         "protrusion_height_mm": max(0.0, max_h * scale_to_mm),

0725: def _get_orthonormal_basis(normal: Vector) -> Tuple[Vector, Vector]:
0726:     """Retorna dois vetores unitários ortogonais à normal fornecida."""
0727:     n = normal.normalized()
0728:     if abs(n.z) < 0.9:

0730:     else:
0731:         u = n.cross(Vector((0.0, 1.0, 0.0))).normalized()

0736: def triangulate_polygon_2d(pts_2d: List[Vector]) -> List[Tuple[int, int, int]]:
0737:     """Triangulação estrita por Ear Clipping para polígono simples 2D (B05 / L01-L06 / C01 / V01).
0738:     
0739:     Garante que 100% dos triângulos fiquem dentro do domínio do polígono,
0740:     inclusive em formas côncavas (L, U, etc.), eliminando triângulos degenerados (área zero)
0741:     mesmo em contornos com vértices colineares intermediários (12, 20, 40+ vértices).
0742:     """
0743:     n = len(pts_2d)
0744:     if n < 3:
0745:         return []
0746:     
0747:     pts_vec = [Vector((p[0], p[1])) if not hasattr(p, 'x') else Vector((p.x, p.y)) for p in pts_2d]
0748: 
0749:     # 1. Área orientada via Shoelace
0750:     area = 0.0
0751:     for i in range(n):
0752:         j = (i + 1) % n
0753:         area += pts_vec[i].x * pts_vec[j].y - pts_vec[j].x * pts_vec[i].y
0754:     area *= 0.5
0755: 
0756:     if abs(area) < 1e-12:
0757:         return []
0758: 
0759:     if n == 3:
0760:         return [(0, 1, 2)] if area > 0 else [(0, 2, 1)]
0761: 
0762:     is_ccw = (area > 0.0)
0763:     indices = list(range(n)) if is_ccw else list(reversed(range(n)))
0764:     poly = [pts_vec[i] for i in indices]
0765: 
0766:     # 2. Identifica cantos essenciais (elimina colineares na mesma reta para o Ear Clipping)
0767:     corners = []
0768:     m_len = len(poly)
0769:     for i in range(m_len):
0770:         p_prev = poly[(i - 1) % m_len]
0771:         p_curr = poly[i]
0772:         p_next = poly[(i + 1) % m_len]
0773: 
0774:         v1 = p_curr - p_prev
0775:         v2 = p_next - p_curr
0776:         l1 = v1.length
0777:         l2 = v2.length
0778: 
0779:         is_collinear = False
0780:         if l1 > 1e-9 and l2 > 1e-9:
0781:             cp = v1.x * v2.y - v1.y * v2.x
0782:             dp = v1.x * v2.x + v1.y * v2.y
0783:             if abs(cp) / (l1 * l2) < 1e-5 and dp > 0:
0784:                 is_collinear = True
0785: 
0786:         if not is_collinear:
0787:             corners.append(i)
0788: 
0789:     if len(corners) < 3:
0790:         raise ValueError("Polígono degenerado: todos os pontos são colineares.")
0791: 
0792:     # 3. Mapeia intermediários colineares pertencentes a cada aresta de canto
0793:     edge_intermediates = {}
0794:     num_corners = len(corners)
0795:     for c_i in range(num_corners):
0796:         c_curr = corners[c_i]
0797:         c_next = corners[(c_i + 1) % num_corners]
0798: 
0799:         inter = []
0800:         if c_next > c_curr:
0801:             for k in range(c_curr + 1, c_next):
0802:                 inter.append(indices[k])
0803:         else:
0804:             for k in range(c_curr + 1, m_len):
0805:                 inter.append(indices[k])
0806:             for k in range(0, c_next):
0807:                 inter.append(indices[k])
0808:         edge_intermediates[(indices[c_curr], indices[c_next])] = inter
0809: 
0810:     # 4. Ear clipping no polígono simplificado de cantos essenciais
0811:     corner_indices = [indices[c] for c in corners]
0812: 
0813:     def is_point_in_tri(pt: Vector, a: Vector, b: Vector, c: Vector) -> bool:
0814:         v0 = c - a
0815:         v1 = b - a
0816:         v2 = pt - a
0817:         dot00 = v0.dot(v0)
0818:         dot01 = v0.dot(v1)
0819:         dot02 = v0.dot(v2)
0820:         dot11 = v1.dot(v1)
0821:         dot12 = v1.dot(v2)
0822:         denom = dot00 * dot11 - dot01 * dot01
0823:         if abs(denom) < 1e-12:
0824:             return False
0825:         inv_denom = 1.0 / denom
0826:         u = (dot11 * dot02 - dot01 * dot12) * inv_denom
0827:         v = (dot00 * dot12 - dot01 * dot02) * inv_denom
0828:         return (u > 1e-5) and (v > 1e-5) and (u + v < 1.0 - 1e-5)
0829: 
0830:     def is_convex(p_prev: Vector, p_curr: Vector, p_next: Vector) -> bool:
0831:         cp = (p_curr.x - p_prev.x) * (p_next.y - p_curr.y) - (p_curr.y - p_prev.y) * (p_next.x - p_curr.x)
0832:         return cp > 1e-8
0833: 
0834:     cur_c_indices = list(corner_indices)
0835:     corner_triangles = []
0836:     it = 0
0837:     max_it = len(cur_c_indices) * len(cur_c_indices) * 2
0838: 
0839:     while len(cur_c_indices) > 3 and it < max_it:
0840:         it += 1
0841:         ear_found = False
0842:         cm = len(cur_c_indices)
0843:         for i in range(cm):
0844:             idx_prev = cur_c_indices[(i - 1) % cm]
0845:             idx_curr = cur_c_indices[i]
0846:             idx_next = cur_c_indices[(i + 1) % cm]
0847: 
0848:             p_prev = pts_vec[idx_prev]
0849:             p_curr = pts_vec[idx_curr]
0850:             p_next = pts_vec[idx_next]
0851: 
0852:             if not is_convex(p_prev, p_curr, p_next):
0853:                 continue
0854: 
0855:             has_pt = False
0856:             for k in range(cm):
0857:                 if k in ((i - 1) % cm, i, (i + 1) % cm):
0858:                     continue
0859:                 pt = pts_vec[cur_c_indices[k]]
0860:                 if is_point_in_tri(pt, p_prev, p_curr, p_next):
0861:                     has_pt = True
0862:                     break
0863: 
0864:             if not has_pt:
0865:                 corner_triangles.append((idx_prev, idx_curr, idx_next))
0866:                 cur_c_indices.pop(i)
0867:                 ear_found = True
0868:                 break
0869: 
0870:         if not ear_found:
0871:             corner_triangles.append((cur_c_indices[0], cur_c_indices[1], cur_c_indices[2]))
0872:             cur_c_indices.pop(1)
0873: 
0874:     if len(cur_c_indices) == 3:
0875:         corner_triangles.append((cur_c_indices[0], cur_c_indices[1], cur_c_indices[2]))
0876: 
0877:     # 5. Subdivide as arestas de cantos que possuem intermediários colineares
0878:     final_triangles = []
0879:     for tri in corner_triangles:
0880:         sub_tris = [tri]
0881:         for e_pair, inter_pts in edge_intermediates.items():
0882:             if not inter_pts:
0883:                 continue
0884:             v_start, v_end = e_pair
0885: 
0886:             new_sub_tris = []
0887:             for st in sub_tris:
0888:                 a, b, c = st
0889:                 found_edge = False
0890:                 for p1, p2, p_opp in ((a, b, c), (b, c, a), (c, a, b)):
0891:                     if (p1 == v_start and p2 == v_end) or (p1 == v_end and p2 == v_start):
0892:                         found_edge = True
0893:                         chain = [p1] + (inter_pts if p1 == v_start else list(reversed(inter_pts))) + [p2]
0894:                         for ch_i in range(len(chain) - 1):
0895:                             new_sub_tris.append((chain[ch_i], chain[ch_i + 1], p_opp))
0896:                         break
0897:                 if not found_edge:
0898:                     new_sub_tris.append(st)
0899:             sub_tris = new_sub_tris
0900: 
0901:         final_triangles.extend(sub_tris)
0902: 
0903:     if not is_ccw:
0904:         final_triangles = [(t[0], t[2], t[1]) for t in final_triangles]
0905: 
0906:     return final_triangles

0909: def create_loop_prism_cutter(
0910:     coords_world: List[Vector],
0911:     inward_dir: Vector,
0912:     depth_bu: float,
0913:     outward_ext_bu: float = 1e-3,
0914:     name: str = "temp_loop_prism",

0916:     """Constrói um prisma 3D fechado e manifold a partir de um contorno de coordenadas mundiais (B05)."""
0917:     bm = bmesh.new()
0918:     in_norm = inward_dir.normalized()
0919: 
0920:     # Remove pontos colineares para evitar triângulos degenerados
0921:     pts = list(coords_world)
0922:     changed = True
0923:     while changed and len(pts) > 3:
0924:         changed = False
0925:         for i in range(len(pts)):
0926:             p_prev, p, p_next = pts[i - 1], pts[i], pts[(i + 1) % len(pts)]
0927:             d1 = p - p_prev
0928:             d2 = p_next - p
0929:             if d1.length < 1e-9 or d2.length < 1e-9 or d1.normalized().cross(d2.normalized()).length < 1e-5:
0930:                 pts.pop(i)
0931:                 changed = True

0934:     n = len(coords_world)
0935: 
0936:     # Projeta contorno para base 2D ortogonal para triangulação Ear Clipping
0937:     u_vec, v_vec = _get_orthonormal_basis(in_norm)
0938:     pts_2d = [Vector((p.dot(u_vec), p.dot(v_vec))) for p in coords_world]
0939:     triangles_2d = triangulate_polygon_2d(pts_2d)
0940: 
0941:     top_verts = [bm.verts.new(p - outward_ext_bu * in_norm) for p in coords_world]
0942:     bot_verts = [bm.verts.new(p + depth_bu * in_norm) for p in coords_world]
0943:     bm.verts.ensure_lookup_table()
0944: 
0945:     # Paredes laterais
0946:     for i in range(n):
0947:         i_next = (i + 1) % n
0948:         bm.faces.new((top_verts[i], top_verts[i_next], bot_verts[i_next], bot_verts[i]))
0949: 
0950:     # Tampas superior e inferior trianguladas estritamente dentro do domínio côncavo (B05)
0951:     for t in triangles_2d:
0952:         # Tampa superior: normal para fora (contra inward_dir)
0953:         bm.faces.new((top_verts[t[2]], top_verts[t[1]], top_verts[t[0]]))
0954:         # Tampa inferior: normal para fora (a favor de inward_dir)
0955:         bm.faces.new((bot_verts[t[0]], bot_verts[t[1]], bot_verts[t[2]]))
0956: 
0957:     bmesh.ops.recalc_face_normals(bm, faces=bm.faces[:])
0958: 
0959:     mesh = bpy.data.meshes.new(f"{name}_Mesh")
0960:     bm.to_mesh(mesh)
0961:     bm.free()
0962: 
0963:     obj = bpy.data.objects.new(name, mesh)
0964:     obj.hide_viewport = False
0965:     obj.hide_render = False
0966:     bpy.context.scene.collection.objects.link(obj)
0967:     return obj

0970: def slice_mesh_by_loop_extrude(
0971:     obj: bpy.types.Object,
0972:     target_name: str,
0973:     indices: List[int],
0974:     seed_face_idx: int,
0975:     depth_mm: float,
0976:     through_cut: bool = False,
0977:     scene: Optional[bpy.types.Scene] = None,
0978: ) -> LoopSliceResult:
0979:     """Corta o modelo extraindo o volume delimitado pelo contorno até a profundidade especificada.
0980:     
0981:     Parte A = volume extraído (prisma ∩ target)
0982:     Parte B = restante do modelo (target \\ prisma)
0983:     """
0984:     actual_scene = scene if scene else bpy.context.scene
0985:     scale_to_mm = get_scene_scale_to_mm(actual_scene)
0986: 
0987:     analysis = analyze_loop_region(obj, indices, seed_face_idx, scene=actual_scene)
0988:     max_avail_mm = analysis["max_avail_depth_mm"]
0989: 
0990:     # Validação de espessura (T09)
0991:     if not analysis.get("thickness_determined", True) and not through_cut:
0992:         raise ValueError(
0993:             "Espessura da peça não pôde ser determinada no ponto selecionado. "
0994:             "Defina a profundidade manualmente ou ative a opção 'Corte Passante'."
0995:         )
0996: 
0997:     if depth_mm > (max_avail_mm + 1e-3) and not through_cut:
0998:         raise ValueError(
0999:             f"A profundidade de extração ({depth_mm:.1f} mm) ultrapassa a espessura da peça "
1000:             f"({max_avail_mm:.1f} mm). Reduza a profundidade ou ative a opção 'Corte Passante'."
1001:         )
1002: 
1003:     coords_world = analysis["coords_world"]
1004:     inward_dir = analysis["inward_dir_world"]
1005:     depth_bu = depth_mm / scale_to_mm
1006: 
1007:     protrusion_h_bu = analysis["protrusion_height_mm"] / scale_to_mm
1008:     outward_ext_bu = max(0.5 / scale_to_mm, protrusion_h_bu + 0.5 / scale_to_mm)
1009: 
1010:     if through_cut and depth_mm >= max_avail_mm:
1011:         depth_bu = max(depth_bu, (max_avail_mm / scale_to_mm) + (1.0 / scale_to_mm))
1012: 
1013:     prism_obj = None
1014:     temp_obj_a = None
1015:     temp_obj_b = None
1016:     eval_obj_a = None
1017:     eval_obj_b = None
1018:     eval_a = None
1019:     eval_b = None
1020: 
1021:     try:
1022:         prism_obj = create_loop_prism_cutter(
1023:             coords_world=coords_world,
1024:             inward_dir=inward_dir,
1025:             depth_bu=depth_bu,
1026:             outward_ext_bu=outward_ext_bu,
1027:             name=f"{target_name}_loop_prism",
1028:         )

1037:         mod_a = temp_obj_a.modifiers.new(name="Extrude_Intersect", type='BOOLEAN')
1038:         mod_a.operation = 'INTERSECT'
1039:         mod_a.solver = 'EXACT'

1043:         temp_obj_b.data = obj.data.copy()
1044:         temp_obj_b.hide_viewport = False
1045:         temp_obj_b.hide_render = False
1046:         actual_scene.collection.objects.link(temp_obj_b)
1047:         mod_b = temp_obj_b.modifiers.new(name="Extrude_Difference", type='BOOLEAN')
1048:         mod_b.operation = 'DIFFERENCE'
1049:         mod_b.solver = 'EXACT'
1050:         mod_b.object = prism_obj
1051: 
1052:         bpy.context.view_layer.update()
1053:         dg = bpy.context.evaluated_depsgraph_get()
1054:

1056:         eval_a = eval_obj_a.to_mesh()
1057: 
1058:         eval_obj_b = temp_obj_b.evaluated_get(dg)
1059:         eval_b = eval_obj_b.to_mesh()

1063:         bm_a.transform(temp_obj_a.matrix_world)
1064: 
1065:         bm_b = bmesh.new()
1066:         bm_b.from_mesh(eval_b)
1067:         bm_b.transform(temp_obj_b.matrix_world)
1068: 
1069:         if len(bm_a.verts) == 0 or len(bm_b.verts) == 0:
1070:             bm_a.free()
1071:             bm_b.free()
1072:             raise RuntimeError("A extração com profundidade gerou malha vazia para uma das partes.")
1073: 
1074:         bmesh.ops.recalc_face_normals(bm_a, faces=bm_a.faces[:])
1075:         bmesh.ops.recalc_face_normals(bm_b, faces=bm_b.faces[:])
1076: 
1077:         interface_origin = analysis["centroid_world"] + (depth_mm / scale_to_mm) * inward_dir
1078:         cut_interface = build_cut_interface(
1079:             origin_world=interface_origin,
1080:             normal_a_to_b=inward_dir,
1081:             boundary_points_world=coords_world,
1082:             unit_scale=scale_to_mm,
1083:             source_mode="LOOP",
1084:         )
1085:

1087:         metrics_b = evaluate_bmesh_metrics(bm_b, unit_scale=scale_to_mm)
1088: 
1089:         split_result = SplitResult(
1090:             part_a_metrics=metrics_a,
1091:             part_b_metrics=metrics_b,
1092:             interface=cut_interface,
1093:             diagnostics_before_connectors={
1094:                 "sum_volume_mm3": metrics_a["volume_mm3"] + metrics_b["volume_mm3"],
1095:                 "interface_width_mm": cut_interface.width_mm,
1096:                 "interface_height_mm": cut_interface.height_mm,
1097:                 "is_planar": True,
1098:                 "deviation_mm": 0.0,
1099:             },
1100:             success=True,
1101:         )
1102: 
1103:         return LoopSliceResult(bm_a, bm_b, split_result)
1104: 
1105:     finally:
1106:         # Liberação estrita no mesmo objeto avaliado que gerou o to_mesh (B03)
1107:         if eval_obj_a and eval_a:

1110:             except Exception:
1111:                 pass

1117: 
1118:         for temp_o in (temp_obj_a, temp_obj_b):
1119:             if temp_o:
1120:                 m_data = temp_o.data
1121:                 if temp_o.name in actual_scene.collection.objects:
1122:                     actual_scene.collection.objects.unlink(temp_o)
1123:                 if temp_o.name in bpy.data.objects:
1124:                     bpy.data.objects.remove(temp_o, do_unlink=True)
1125:                 if m_data and m_data.users == 0:
1126:                     bpy.data.meshes.remove(m_data, do_unlink=True)
