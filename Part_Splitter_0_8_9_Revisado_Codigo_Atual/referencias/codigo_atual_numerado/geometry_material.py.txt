0022: def _get_orthonormal_basis(normal: Vector) -> Tuple[Vector, Vector]:
0023:     """Retorna dois vetores unitários ortogonais à normal fornecida."""
0024:     n = normal.normalized()
0025:     if abs(n.z) < 0.9:
0026:         u = n.cross(Vector((0.0, 0.0, 1.0))).normalized()
0027:     else:
0028:         u = n.cross(Vector((0.0, 1.0, 0.0))).normalized()
0029:     v = n.cross(u).normalized()
0030:     return u, v

0033: def get_material_regions(bm, material_index: int) -> List[List[int]]:
0034:     """Agrupa as faces de um material em regiões conectadas (ilhas) por arestas compartilhadas.
0035: 
0036:     Retorna listas de índices de faces, ordenadas pelo menor índice de face (ordem estável).
0037:     """
0038:     bm.faces.ensure_lookup_table()
0039:     visited = set()
0040:     regions: List[List[int]] = []
0041:     for f in bm.faces:
0042:         if f.material_index != material_index or f.index in visited:
0043:             continue
0044:         stack = [f]
0045:         visited.add(f.index)
0046:         region = []
0047:         while stack:
0048:             cur = stack.pop()
0049:             region.append(cur.index)
0050:             for e in cur.edges:
0051:                 for nf in e.link_faces:
0052:                     if nf.material_index == material_index and nf.index not in visited:
0053:                         visited.add(nf.index)
0054:                         stack.append(nf)
0055:         regions.append(sorted(region))
0056:     regions.sort(key=lambda r: r[0])
0057:     return regions

0060: def _order_boundary_loops(boundary_edges):
0061:     """Ordena arestas de fronteira em um ou mais ciclos fechados de vértices.
0062: 
0063:     Retorna None se houver bifurcações (vértice com número de arestas diferente de 2).
0064:     """
0065:     vert_to_edges: Dict[Any, List[Any]] = {}
0066:     for e in boundary_edges:
0067:         for v in e.verts:
0068:             vert_to_edges.setdefault(v, []).append(e)
0069:     if not vert_to_edges or any(len(es) != 2 for es in vert_to_edges.values()):
0070:         return None
0071: 
0072:     loops = []
0073:     used_edges = set()
0074:     for start_v in list(vert_to_edges.keys()):
0075:         if all(e in used_edges for e in vert_to_edges[start_v]):
0076:             continue
0077:         ordered = [start_v]
0078:         curr_v = start_v
0079:         curr_e = next(e for e in vert_to_edges[start_v] if e not in used_edges)
0080:         while True:
0081:             used_edges.add(curr_e)
0082:             next_v = curr_e.other_vert(curr_v)
0083:             if next_v == start_v:
0084:                 break
0085:             ordered.append(next_v)
0086:             es = vert_to_edges[next_v]
0087:             curr_e = es[0] if es[1] == curr_e else es[1]
0088:             curr_v = next_v
0089:         loops.append(ordered)
0090:     return loops

0119: def analyze_material_structure(obj: bpy.types.Object, material_index: int, region_index: int = 0) -> Dict[str, Any]:
0120:     """Analisa a topologia de UMA região conectada (ilha) de faces de um material.
0121: 
0122:     Um mesmo material pode aparecer em várias áreas separadas do modelo (ex: dois quadrados
0123:     rosa em lados diferentes). Cada área é tratada como uma região independente, com seu
0124:     próprio contorno e direção de extração.
0125:     """
0126:     if not obj or obj.type != 'MESH':
0127:         return {"valid": False, "error": "Objeto inválido."}
0128: 
0129:     bm = bmesh.new()
0130:     bm.from_mesh(obj.data)
0131:     bm.transform(obj.matrix_world)
0132:     bm.verts.ensure_lookup_table()
0133:     bm.edges.ensure_lookup_table()
0134:     bm.faces.ensure_lookup_table()
0135: 
0136:     total_faces = len(bm.faces)
0137:     regions = get_material_regions(bm, material_index)
0138: 
0139:     if not regions:
0140:         bm.free()
0141:         return {"valid": False, "error": f"Nenhuma face com o material #{material_index} encontrada.", "region_count": 0}
0142: 
0143:     region_count = len(regions)
0144:     region_index = max(0, min(region_index, region_count - 1))
0145:     region_faces = set(regions[region_index])
0146: 
0147:     if len(region_faces) == total_faces:
0148:         bm.free()
0149:         return {"valid": False, "error": "Todas as faces do modelo já utilizam este material.", "region_count": region_count}
0150: 
0151:     target_faces = [bm.faces[i] for i in region_faces]
0152: 
0153:     # Arestas de fronteira da região: uma face dentro e outra fora da região
0154:     boundary_edges = []
0155:     for e in bm.edges:
0156:         inside = [f for f in e.link_faces if f.index in region_faces]
0157:         if inside and len(inside) != len(e.link_faces):
0158:             boundary_edges.append(e)
0159: 
0160:     base_info = {
0161:         "region_count": region_count,
0162:         "region_index": region_index,
0163:         "region_face_indices": sorted(region_faces),
0164:         "face_count": len(region_faces),
0165:     }
0166: 
0167:     # Caso 1: região sem fronteira com outras faces -> componente fechado isolado
0168:     if not boundary_edges:
0169:         open_edges = [e for e in bm.edges
0170:                       if any(f.index in region_faces for f in e.link_faces) and not e.is_manifold]
0171:         is_manifold = len(open_edges) == 0
0172:         bm.free()
0173:         info = dict(base_info)
0174:         info.update({
0175:             "valid": True,
0176:             "type": "CLOSED_COMPONENT",
0177:             "description": "Componente 3D Fechado (Sólido estanque)",
0178:             "is_manifold": is_manifold,
0179:             "boundary_loops": [],
0180:         })
0181:         return info
0182: 
0183:     # Caso 2: região superficial com contorno
0184:     loops = _order_boundary_loops(boundary_edges)
0185:     if loops is None:
0186:         bm.free()
0187:         info = dict(base_info)
0188:         info.update({
0189:             "valid": False,
0190:             "type": "COMPLEX_BOUNDARY",
0191:             "error": "O contorno desta região tem bifurcações (cantos tocando outras regiões).",
0192:         })
0193:         return info
0194: 
0195:     if len(loops) > 1:
0196:         bm.free()
0197:         info = dict(base_info)
0198:         info.update({
0199:             "valid": False,
0200:             "type": "BASE_REGION",
0201:             "error": "Esta região é o corpo/base (envolve outras áreas). Escolha a cor da área menor.",
0202:         })
0203:         return info
0204: 
0205:     ordered_verts = loops[0]
0206:     coords_world = [v.co.copy() for v in ordered_verts]
0207:     centroid = sum(coords_world, Vector()) / len(coords_world)
0208: 
0209:     # Normal média do contorno via Newell
0210:     n = Vector((0.0, 0.0, 0.0))
0211:     n_pts = len(coords_world)
0212:     for i in range(n_pts):
0213:         v1 = coords_world[i]
0214:         v2 = coords_world[(i + 1) % n_pts]
0215:         n.x += (v1.y - v2.y) * (v1.z + v2.z)
0216:         n.y += (v1.z - v2.z) * (v1.x + v2.x)
0217:         n.z += (v1.x - v2.x) * (v1.y + v2.y)
0218:     if n.length > 1e-6:
0219:         n.normalize()
0220:     else:
0221:         n = Vector((0.0, 0.0, 1.0))
0222: 
0223:     # Direção para dentro a partir da média ponderada por área em espaço de mundo
0224:     avg_norm = sum((f.normal * f.calc_area() for f in target_faces), Vector())
0225:     if avg_norm.length > 1e-9:
0226:         avg_norm.normalize()
0227:     else:
0228:         avg_norm = n
0229:     inward_dir = -avg_norm
0230: 
0231:     # Material adjacente predominante fora da região (corpo receptor)
0232:     adj_outside_mats = []
0233:     for e in boundary_edges:
0234:         for f in e.link_faces:
0235:             if f.index not in region_faces:
0236:                 adj_outside_mats.append(f.material_index)
0237:     body_mat_index = max(set(adj_outside_mats), key=adj_outside_mats.count) if adj_outside_mats else 0
0238: 
0239:     scale_to_mm = get_scene_scale_to_mm(bpy.context.scene)
0240:     dists = [abs((p - centroid).dot(n)) for p in coords_world]
0241:     max_dev_mm = (max(dists) if dists else 0.0) * scale_to_mm
0242:     is_planar_base = (max_dev_mm <= 0.15)
0243: 
0244:     # Classificação de faces: saliência volumétrica vs faces coplanares da base
0245:     tol_planar_bu = 0.15 / scale_to_mm
0246:     protrusion_face_indices = []
0247:     coplanar_base_face_indices = []
0248:     for f in target_faces:
0249:         face_dists = [abs((v.co - centroid).dot(n)) for v in f.verts]
0250:         if max(face_dists) <= tol_planar_bu and abs(abs(f.normal.dot(n)) - 1.0) <= 0.05:
0251:             coplanar_base_face_indices.append(f.index)
0252:         else:
0253:             protrusion_face_indices.append(f.index)
0254: 
0255:     bm.free()
0256: 
0257:     info = dict(base_info)
0258:     info.update({
0259:         "valid": True,
0260:         "type": "SURFACE_PATCH",
0261:         "description": "Fronteira na Superfície (Requer profundidade de extração)",
0262:         "coords_world": coords_world,
0263:         "centroid_world": centroid,
0264:         "normal_world": n,
0265:         "inward_dir_world": inward_dir,
0266:         "is_planar_base": is_planar_base,
0267:         "base_deviation_mm": max_dev_mm,
0268:         "base_normal_world": n,
0269:         "body_mat_index": body_mat_index,
0270:         "protrusion_face_indices": protrusion_face_indices,
0271:         "coplanar_base_face_indices": coplanar_base_face_indices,
0272:     })
0273:     return info

0276: def create_material_patch_prism_cutter(
0277:     obj: bpy.types.Object,
0278:     region_face_indices: List[int],
0279:     inward_dir: Vector,
0280:     depth_bu: float,
0281:     outward_ext_bu: float = 1e-3,
0282:     name: str = "temp_mat_prism",
0283:     material_index: int = 0,
0284:     interface_mat_mode: str = 'SOURCE',
0285:     body_mat_index: int = 0,
0286: ) -> bpy.types.Object:
0287:     """Constrói um prisma 3D estanque a partir das faces exatas da região de material (B06).
0288:     
0289:     Preserva a tesselação e concavidades reais da região (ex: silhuetas em U ou reentrâncias),
0290:     evitando tampas em n-gons planos que cruzam aberturas e geram películas/membranas espúrias.
0291:     Valida a estanqueidade e ausência de degeneração antes da publicação (D01).
0292:     """
0293:     bm_src = bmesh.new()
0294:     bm_src.from_mesh(obj.data)
0295:     bm_src.transform(obj.matrix_world)
0296:     bm_src.verts.ensure_lookup_table()
0297:     bm_src.edges.ensure_lookup_table()
0298:     bm_src.faces.ensure_lookup_table()
0299: 
0300:     region_set = set(region_face_indices)
0301:     region_faces = [bm_src.faces[i] for i in region_face_indices if i < len(bm_src.faces)]
0302: 
0303:     in_norm = inward_dir.normalized()
0304:     bm_prism = bmesh.new()
0305: 
0306:     top_map = {}
0307:     bot_map = {}
0308:     used_verts = {v for f in region_faces for v in f.verts}
0309: 
0310:     for v in used_verts:
0311:         top_map[v] = bm_prism.verts.new(v.co - outward_ext_bu * in_norm)
0312:         bot_map[v] = bm_prism.verts.new(v.co + depth_bu * in_norm)
0313: 
0314:     bm_prism.verts.ensure_lookup_table()
0315: 
0316:     target_iface_mat = material_index if interface_mat_mode == 'EXTRACTED' else body_mat_index
0317: 
0318:     # Tampas superior e inferior preservando a tesselação exata das faces
0319:     for f in region_faces:
0320:         f_top = bm_prism.faces.new([top_map[v] for v in f.verts])
0321:         f_top.material_index = material_index
0322:         f_bot = bm_prism.faces.new(reversed([bot_map[v] for v in f.verts]))
0323:         f_bot.material_index = target_iface_mat
0324: 
0325:     # Paredes laterais apenas nas arestas de fronteira da região
0326:     for e in bm_src.edges:
0327:         inc = [f for f in e.link_faces if f.index in region_set]
0328:         if len(inc) == 1:
0329:             f = inc[0]
0330:             v1, v2 = e.verts[0], e.verts[1]
0331:             f_v = list(f.verts)
0332:             idx1 = f_v.index(v1)
0333:             if f_v[(idx1 + 1) % len(f_v)] == v2:
0334:                 va, vb = v1, v2
0335:             else:
0336:                 va, vb = v2, v1
0337:             f_side = bm_prism.faces.new((top_map[vb], top_map[va], bot_map[va], bot_map[vb]))
0338:             f_side.material_index = target_iface_mat
0339: 
0340:     bm_src.free()
0341: 
0342:     bmesh.ops.triangulate(bm_prism, faces=bm_prism.faces[:])
0343:     bmesh.ops.recalc_face_normals(bm_prism, faces=bm_prism.faces[:])
0344: 
0345:     # Validação do cortador antes de qualquer operação booleana (D01)
0346:     scale_to_mm = get_scene_scale_to_mm(bpy.context.scene)
0347:     metrics = evaluate_bmesh_metrics(bm_prism, scale_to_mm)
0348:     if not metrics["is_manifold"] or metrics["volume_mm3"] <= 1e-4:
0349:         bm_prism.free()
0350:         raise ValueError(
0351:             "O cortador prismático gerou geometria degenerada ou não-estanque (auto-interseção por faces dobradas no patch de material). "
0352:             "A extração prismática requer uma superfície com projeção desimpedida na direção de extração."
0353:         )
0354: 
0355:     mesh = bpy.data.meshes.new(f"{name}_Mesh")
0356:     # Copia os slots de material do objeto original para o cortador evitar faces com slot 0 genérico
0357:     for mat in obj.data.materials:
0358:         mesh.materials.append(mat)
0359: 
0360:     bm_prism.to_mesh(mesh)
0361:     bm_prism.free()
0362: 
0363:     prism_obj = bpy.data.objects.new(name, mesh)
0364:     prism_obj.hide_viewport = False
0365:     prism_obj.hide_render = False
0366:     bpy.context.scene.collection.objects.link(prism_obj)
0367:     return prism_obj

0370: def slice_mesh_by_material(
0371:     obj: bpy.types.Object,
0372:     material_index: int,
0373:     depth_mm: float = 5.0,
0374:     scene: Optional[bpy.types.Scene] = None,
0375:     through_cut: bool = False,
0376:     region_index: int = 0,
0377:     extraction_mode: str = 'EXTRUDE',

0383:     
0384:     Suporta os modos:
0385:     - 'BASE': Destaca a saliência volumétrica inteira pela base com tampas planas (C01),
0386:               conservação estrita de volume (C02) e alívio dimensional calibrado (C03).
0387:     - 'EXTRUDE': Extração prismática superficial com profundidade definida e cavidade complementar (M03).
0388:     """
0389:     actual_scene = scene if scene else bpy.context.scene
0390:     scale_to_mm = get_scene_scale_to_mm(actual_scene)
0391: 
0392:     analysis = analyze_material_structure(obj, material_index, region_index)
0393:     if not analysis["valid"]:
0394:         raise ValueError(analysis.get("error", "Estrutura de material não suportada."))
0395: 
0396:     mat_type = analysis["type"]
0397: 
0398:     if mat_type == "CLOSED_COMPONENT":
0399:         region_set = set(analysis["region_face_indices"])
0400:         # Separação direta de componente estanque já fechado
0401:         bm = bmesh.new()
0402:         bm.from_mesh(obj.data)
0403:         bm.transform(obj.matrix_world)
0404: 
0405:         bm_a = bm.copy()
0406:         bm_a.faces.ensure_lookup_table()
0407:         bmesh.ops.delete(bm_a, geom=[f for f in bm_a.faces if f.index not in region_set], context='FACES')
0408: 
0409:         bm_b = bm.copy()
0410:         bm_b.faces.ensure_lookup_table()
0411:         bmesh.ops.delete(bm_b, geom=[f for f in bm_b.faces if f.index in region_set], context='FACES')
0412: 
0413:         bm.free()
0414: 
0415:         # Direção de separação A->B: do centro do componente para o centro do corpo
0416:         ca = sum((v.co for v in bm_a.verts), Vector()) / max(1, len(bm_a.verts))
0417:         cb = sum((v.co for v in bm_b.verts), Vector()) / max(1, len(bm_b.verts))
0418:         sep = cb - ca
0419:         sep_norm = sep.normalized() if sep.length > 1e-9 else Vector((0.0, 0.0, -1.0))
0420: 
0421:         cut_iface = CutInterface(
0422:             origin_world=ca.copy(),
0423:             normal_a_to_b_world=sep_norm,
0424:             axis_u_world=Vector((1.0, 0.0, 0.0)),
0425:             axis_v_world=Vector((0.0, 1.0, 0.0)),
0426:             outer_contours_uv=[],
0427:             hole_contours_uv=[],
0428:             interface_triangles_world=[],
0429:             is_planar=True,
0430:             deviation_mm=0.0,
0431:             source_mode='MATERIAL',
0432:         )
0433: 
0434:         metrics_a = evaluate_bmesh_metrics(bm_a, scale_to_mm)
0435:         metrics_b = evaluate_bmesh_metrics(bm_b, scale_to_mm)
0436: 
0437:         split_res = SplitResult(
0438:             part_a_metrics=metrics_a,
0439:             part_b_metrics=metrics_b,
0440:             interface=cut_iface,
0441:             diagnostics_before_connectors={
0442:                 "part_a_volume_mm3": metrics_a["volume_mm3"],
0443:                 "part_b_volume_mm3": metrics_b["volume_mm3"],
0444:                 "sum_volume_mm3": metrics_a["volume_mm3"] + metrics_b["volume_mm3"],
0445:             },
0446:             success=True,
0447:         )
0448:         return LoopSliceResult(bm_a, bm_b, split_res)
0449: 
0450:     elif extraction_mode == 'BASE':
0451:         # C01 / C02 / C03: Destacar saliência pela base com fechamento planar e conservação volumétrica
0452:         if not analysis.get("is_planar_base", True):
0453:             raise ValueError(
0454:                 f"A base da saliência não é estritamente planar (desvio de planicidade: "
0455:                 f"{analysis.get('base_deviation_mm', 0.0):.2f} mm > 0.15 mm). "
0456:                 "Pinte todas as faces da saliência até sua ligação com o corpo ou utilize o modo Loop de Arestas."
0457:             )
0458: 
0459:         protrusion_indices = analysis.get("protrusion_face_indices")
0460:         coplanar_indices = analysis.get("coplanar_base_face_indices", [])
0461: 
0462:         # Se a região contém saliência E faces coplanares da base, extrai a saliência e mantém as faces coplanares no corpo (C01/C02)
0463:         if protrusion_indices and len(protrusion_indices) > 0:
0464:             target_indices = protrusion_indices
0465:         elif not protrusion_indices and len(coplanar_indices) > 0:
0466:             raise ValueError(
0467:                 "A região de material selecionada é completamente coplanar com a base (não possui saliência volumétrica). "
0468:                 "Para recortar superfícies planas, utilize o modo 'Extrair com Profundidade'."
0469:             )
0470:         else:
0471:             target_indices = analysis["region_face_indices"]
0472: 
0473:         region_set = set(target_indices)
0474:         bm = bmesh.new()
