0019: def _get_orthonormal_basis(normal: Vector) -> Tuple[Vector, Vector]:
0020:     """Retorna dois vetores unitários ortogonais à normal fornecida."""
0021:     n = normal.normalized()
0022:     if abs(n.z) < 0.9:
0023:         u = n.cross(Vector((0.0, 0.0, 1.0))).normalized()
0024:     else:
0025:         u = n.cross(Vector((0.0, 1.0, 0.0))).normalized()
0026:     v = n.cross(u).normalized()
0027:     return u, v

0135: def _compute_cut_line_segments(mesh_or_bm, matrix_world: Matrix, plane_co: Vector, plane_no: Vector, is_bmesh: bool = False) -> List[Vector]:
0136:     """Calcula segmentos de linha estritos da interseção entre o plano e cada triângulo/face da malha (C04 / C12 / V14).
0137:     
0138:     Garante que cada segmento pertença a uma única face, eliminando traços espúrios que cortam o vazio
0139:     e detectando com precisão arestas coincidentes ou tangentes ao plano de corte.
0140:     """
0141:     segments = []
0142:     norm = plane_no.normalized()
0143: 
0144:     if is_bmesh:
0145:         bm = mesh_or_bm
0146:         for f in bm.faces:
0147:             if not f.is_valid or len(f.verts) < 3:
0148:                 continue
0149:             v_coords = [matrix_world @ v.co for v in f.verts]
0150:             n_v = len(v_coords)
0151:             for i in range(1, n_v - 1):
0152:                 tri = (v_coords[0], v_coords[i], v_coords[i + 1])
0153:                 dists = [(p - plane_co).dot(norm) for p in tri]
0154:                 on_plane = [abs(d) <= 1e-6 for d in dists]
0155:                 num_on_plane = sum(on_plane)
0156:                 has_pos = any(d > 1e-6 for d in dists)
0157:                 has_neg = any(d < -1e-6 for d in dists)
0158: 
0159:                 if not (has_pos and has_neg) and num_on_plane < 2:
0160:                     continue
0161: 
0162:                 inter_pts = []
0163:                 if num_on_plane == 2:
0164:                     for v_idx in range(3):
0165:                         if on_plane[v_idx]:
0166:                             inter_pts.append(tri[v_idx])
0167:                 elif num_on_plane == 3:
0168:                     continue
0169:                 else:
0170:                     for e_idx in range(3):
0171:                         p_a = tri[e_idx]
0172:                         p_b = tri[(e_idx + 1) % 3]
0173:                         d_a = dists[e_idx]
0174:                         d_b = dists[(e_idx + 1) % 3]
0175:                         if (d_a > 1e-6 and d_b < -1e-6) or (d_a < -1e-6 and d_b > 1e-6):
0176:                             t = d_a / (d_a - d_b)
0177:                             inter_pts.append(p_a + t * (p_b - p_a))
0178:                         elif abs(d_a) <= 1e-6 and not any((p_a - ip).length_squared < 1e-10 for ip in inter_pts):
0179:                             inter_pts.append(p_a)
0180: 
0181:                 if len(inter_pts) == 2:
0182:                     segments.append(inter_pts[0])
0183:                     segments.append(inter_pts[1])
0184:     else:
0185:         mesh = mesh_or_bm
0186:         if not mesh or not mesh.polygons or not mesh.vertices:
0187:             return []
0188:         mw = matrix_world
0189:         verts_co = [mw @ v.co for v in mesh.vertices]
0190:         for poly in mesh.polygons:
0191:             p_verts = poly.vertices
0192:             n_v = len(p_verts)
0193:             if n_v < 3:
0194:                 continue
0195:             v0_co = verts_co[p_verts[0]]
0196:             d0 = (v0_co - plane_co).dot(norm)
0197:             for i in range(1, n_v - 1):
0198:                 v1_co = verts_co[p_verts[i]]
0199:                 v2_co = verts_co[p_verts[i + 1]]
0200:                 d1 = (v1_co - plane_co).dot(norm)
0201:                 d2 = (v2_co - plane_co).dot(norm)
0202: 
0203:                 dists = (d0, d1, d2)
0204:                 on_plane = (abs(d0) <= 1e-6, abs(d1) <= 1e-6, abs(d2) <= 1e-6)
0205:                 num_on_plane = sum(on_plane)
0206:                 has_pos = any(d > 1e-6 for d in dists)
0207:                 has_neg = any(d < -1e-6 for d in dists)
0208: 
0209:                 if not (has_pos and has_neg) and num_on_plane < 2:
0210:                     continue
0211: 
0212:                 tri = (v0_co, v1_co, v2_co)
0213:                 inter_pts = []
0214:                 if num_on_plane == 2:
0215:                     for v_idx in range(3):
0216:                         if on_plane[v_idx]:
0217:                             inter_pts.append(tri[v_idx])
0218:                 elif num_on_plane == 3:
0219:                     continue
0220:                 else:
0221:                     for e_idx in range(3):
0222:                         p_a = tri[e_idx]
0223:                         p_b = tri[(e_idx + 1) % 3]
0224:                         d_a = dists[e_idx]
0225:                         d_b = dists[(e_idx + 1) % 3]
0226:                         if (d_a > 1e-6 and d_b < -1e-6) or (d_a < -1e-6 and d_b > 1e-6):
0227:                             t = d_a / (d_a - d_b)
0228:                             inter_pts.append(p_a + t * (p_b - p_a))
0229:                         elif abs(d_a) <= 1e-6 and not any((p_a - ip).length_squared < 1e-10 for ip in inter_pts):
0230:                             inter_pts.append(p_a)
0231: 
0232:                 if len(inter_pts) == 2:
0233:                     segments.append(inter_pts[0])
0234:                     segments.append(inter_pts[1])
0235: 
0236:     return segments
