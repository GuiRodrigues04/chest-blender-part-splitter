0023: def sanitize_unique_filename(name: str, used_names: Set[str], ext: str = ".stl") -> str:
0024:     """Sanitiza um nome de arquivo e garante unicidade evitando colisões ou sobrescrita silenciosa."""
0025:     clean = "".join(c for c in name if c.isalnum() or c in ('_', '-'))
0026:     clean = clean.strip("._-")
0027:     if not clean:
0028:         clean = "Part"
0029:     
0030:     candidate = clean
0031:     counter = 1
0032:     while candidate.lower() in used_names:
0033:         counter += 1
0034:         candidate = f"{clean}_{counter}"
0035:     used_names.add(candidate.lower())
0036:     return f"{candidate}{ext}"

0041: def get_material_rgba(mat: bpy.types.Material) -> Tuple[float, float, float, float]:
0042:     """Extrai cor RGBA de um material do Blender (diffuse_color ou Principled BSDF)."""
0043:     if not mat:
0044:         return (0.8, 0.8, 0.8, 1.0)
0045:     
0046:     # 1. Tenta obter do nó Principled BSDF
0047:     if mat.use_nodes and mat.node_tree:
0048:         for node in mat.node_tree.nodes:
0049:             if node.type == 'BSDF_PRINCIPLED':
0050:                 base_col_sock = node.inputs.get("Base Color")
0051:                 if base_col_sock and hasattr(base_col_sock, "default_value"):
0052:                     col = base_col_sock.default_value
0053:                     return (col[0], col[1], col[2], col[3] if len(col) > 3 else 1.0)
0054: 
0055:     # 2. Fallback para diffuse_color
0056:     d = mat.diffuse_color
0057:     return (d[0], d[1], d[2], d[3] if len(d) > 3 else 1.0)

0060: def rgba_to_hex(rgba: Tuple[float, float, float, float]) -> str:
0061:     """Converte RGBA float (0..1) para string hexadecimal '#RRGGBBAA'."""
0062:     r = min(255, max(0, int(rgba[0] * 255)))
0063:     g = min(255, max(0, int(rgba[1] * 255)))
0064:     b = min(255, max(0, int(rgba[2] * 255)))
0065:     a = min(255, max(0, int(rgba[3] * 255)))
0066:     return f"#{r:02X}{g:02X}{b:02X}{a:02X}"

0069: def publish_staged_files_atomic(staged_files: List[Tuple[str, str]], target_dir: str) -> List[str]:
0070:     """Publica arquivos estagiados para o diretório de destino de forma atômica com rollback.
0071:     
0072:     Se ocorrer qualquer falha durante a publicação de qualquer arquivo:
0073:     - Todos os arquivos recém-criados no destino durante esta operação são removidos.
0074:     - Todos os arquivos preexistentes que foram substituídos são restaurados de seus backups.
0075:     - Nenhuma saída parcial ou corrompida permanece.
0076:     """
0077:     os.makedirs(target_dir, exist_ok=True)
0078:     backup_dir = tempfile.mkdtemp(prefix="chest_publish_backup_")
0079:     copied_new: List[str] = []
0080:     replaced_backups: List[Tuple[str, str]] = []  # (target_path, backup_path)
0081:     published_files: List[str] = []
0082: 
0083:     try:
0084:         for stage_path, filename in staged_files:
0085:             target_path = os.path.join(target_dir, filename)
0086:             if os.path.exists(target_path):
0087:                 backup_path = os.path.join(backup_dir, f"bak_{len(replaced_backups)}_{filename}")
0088:                 shutil.copy2(target_path, backup_path)
0089:                 replaced_backups.append((target_path, backup_path))
0090:             else:
0091:                 copied_new.append(target_path)
0092:             
0093:             shutil.copy2(stage_path, target_path)
0094:             published_files.append(target_path)
0095:             
0096:     except Exception as e:
0097:         for p in copied_new:
0098:             if os.path.exists(p):
0099:                 try:
0100:                     os.remove(p)
0101:                 except OSError:
0102:                     pass
0103:         for target_path, backup_path in replaced_backups:
0104:             if os.path.exists(backup_path):
0105:                 try:
0106:                     shutil.copy2(backup_path, target_path)
0107:                 except OSError:
0108:                     pass
0109:         raise RuntimeError(f"Falha durante a publicação dos arquivos no destino: {e}. Rollback executado com sucesso.") from e
0110:     finally:
0111:         if os.path.exists(backup_dir):
0112:             try:
0113:                 shutil.rmtree(backup_dir)
0114:             except OSError:
0115:                 pass
0116: 
0117:     return published_files

0184:         bm = bmesh.new()
0185:         eval_obj = None
0186:         eval_mesh = None

0220:             v1 = f.verts[1].index

0222:             
0223:             c_idx = 0
0224:             if f.material_index < len(mat_slots):
0225:                 m_slot = mat_slots[f.material_index]

0290: def separate_object_by_material(source_obj: bpy.types.Object, target_collection: bpy.types.Collection) -> List[bpy.types.Object]:
0291:     """Divide um objeto malha em múltiplos sub-objetos individuais, um para cada material/cor.
0292:     
0293:     Cada sub-objeto retém suas faces correspondentes, seu material exclusivo e a mesma
0294:     origem de coordenadas de mundo, permitindo importação e alinhamento direto no fatiador.
0295:     """
0296:     if source_obj.type != 'MESH' or not source_obj.data:
0297:         return []
0298: 
0299:     materials = source_obj.data.materials
0300:     if len(materials) <= 1:
0301:         return [source_obj]
0302: 
0303:     created_objs = []
0304:     base_name = source_obj.name
0305: 
0306:     for mat_idx, mat in enumerate(materials):
0307:         mat_name = mat.name if mat else f"Mat_{mat_idx}"
0308:         
0309:         bm = bmesh.new()
0310:         bm.from_mesh(source_obj.data)
0311:         bm.faces.ensure_lookup_table()
0312:         
0313:         faces_to_delete = [f for f in bm.faces if f.material_index != mat_idx]
0314:         if len(faces_to_delete) == len(bm.faces):
0315:             bm.free()
0316:             continue
0317: 
0318:         bmesh.ops.delete(bm, geom=faces_to_delete, context='FACES')
0319: 
0320:         sub_mesh = bpy.data.meshes.new(f"{base_name}__{mat_name}")
0321:         if mat:
0322:             sub_mesh.materials.append(mat)
0323:         
0324:         for f in bm.faces:
0325:             f.material_index = 0
0326:             
0327:         bm.to_mesh(sub_mesh)
0328:         sub_mesh.update()
0329:         bm.free()
0330: 
0331:         sub_obj = bpy.data.objects.new(f"{base_name}__{mat_name}", sub_mesh)
0332:         sub_obj.matrix_world = source_obj.matrix_world.copy()
0333:         if mat:
0334:             sub_obj.color = mat.diffuse_color
0335: 
0336:         # Validação estrita de contrato de sólidos para impressão 3D (Seção 9)
0337:         bm_check = bmesh.new()
0338:         bm_check.from_mesh(sub_mesh)
0339:         is_manifold = all(e.is_manifold for e in bm_check.edges)
0340:         open_edges = len([e for e in bm_check.edges if not e.is_manifold])
0341:         bm_check.free()
0342: 
0343:         sub_obj["is_manifold"] = is_manifold
0344:         sub_obj["open_edges"] = open_edges
0345:         sub_obj["is_solid_for_printing"] = is_manifold
0346:         sub_obj["chest_session_id"] = source_obj.get("chest_session_id", source_obj.get("chest_splitter_session_id", ""))
0347:         sub_obj["chest_role"] = "SUBPART_MATERIAL"
0348:         if "chest_splitter_part" in source_obj:
0349:             sub_obj["chest_splitter_part"] = source_obj["chest_splitter_part"]
0350: 
0351:         target_collection.objects.link(sub_obj)
0352:         created_objs.append(sub_obj)
0353: 
0354:     return created_objs
