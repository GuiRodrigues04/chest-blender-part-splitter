0135: def capture_target_snapshot(scene: bpy.types.Scene, op_record: Any, target: bpy.types.Object):
0136:     """Cria ou atualiza transacionalmente o snapshot do alvo antes de ocultá-lo (B13)."""
0137:     if not op_record or not target or target.type != 'MESH':
0138:         return
0139: 
0140:     backups_col = ensure_backups_collection(scene)
0141:     op_id = op_record.operation_id
0142: 
0143:     # 1. Cria primeiro o candidato independente com integridade total (B13 / T40)
0144:     snap_mesh = target.data.copy()
0145:     snap_mesh.name = f"_snap_mesh_{op_id[:8]}"
0146: 
0147:     snap_obj = target.copy()
0148:     snap_obj.data = snap_mesh
0149:     snap_obj.name = f"_snap_obj_{op_id[:8]}"
0150:     snap_obj["chest_role"] = ROLE_RESTORE_BACKUP
0151:     snap_obj["chest_operation_id"] = op_id
0152: 
0153:     backups_col.objects.link(snap_obj)
0154: 
0155:     # 2. Somente após garantir a criação do novo candidato, descarta o snapshot anterior
0156:     old_snap_obj = op_record.snapshot_object
0157:     if old_snap_obj and old_snap_obj != snap_obj:
0158:         discard_target_snapshot(scene, op_record)
0159: 
0160:     # 3. Promove o novo snapshot validado
0161:     op_record.snapshot_object = snap_obj
0162:     op_record.snapshot_name = snap_obj.name
0163:     op_record.target_object = target
0164:     op_record.target_name = target.name
0165:     op_record.target_hide_viewport = target.hide_viewport
0166:     op_record.target_hide_render = target.hide_render

0169: def discard_target_snapshot(scene: bpy.types.Scene, op_record: Any):
0170:     """Descarta o snapshot de backup associado a uma operação de prévia com validação estrita (T42)."""
0171:     if not op_record:
0172:         return
0173:     snap_obj = op_record.snapshot_object
0174:     snap_name = op_record.snapshot_name
0175: 
0176:     backups_col = find_collection(scene, COLLECTION_BACKUPS_NAME)
0177: 
0178:     target_snap = snap_obj if (snap_obj and snap_obj.name in bpy.data.objects) else bpy.data.objects.get(snap_name)
0179:     if target_snap:
0180:         # Validação estrita: só descartar se for realmente o snapshot desta operação (T42)
0181:         if (target_snap.get("chest_role") == ROLE_RESTORE_BACKUP and
0182:             target_snap.get("chest_operation_id") == op_record.operation_id):
0183:             snap_mesh = target_snap.data
0184:             if backups_col and target_snap.name in backups_col.objects:
0185:                 backups_col.objects.unlink(target_snap)
0186:             if target_snap.name in bpy.data.objects:
0187:                 bpy.data.objects.remove(target_snap, do_unlink=True)
0188:             if snap_mesh and snap_mesh.users == 0:
0189:                 bpy.data.meshes.remove(snap_mesh, do_unlink=True)
0190: 
0191:     op_record.snapshot_object = None
0192:     op_record.snapshot_name = ""

0199: def resolve_operation_status(context: bpy.types.Context) -> Tuple[Optional[Any], str]:
0200:     """Identifica de forma determinística e estruturada a operação em foco e seu status (C19 / V21).
0201:     
0202:     Retorna (op_record, status_code) onde status_code pode ser:
0203:     - 'FOUND': Operação unívoca encontrada
0204:     - 'NOT_FOUND': Nenhuma operação encontrada
0205:     - 'AMBIGUOUS': Conflito de seleção entre múltiplas operações
0206:     """
0207:     scene = context.scene
0208:     settings = getattr(scene, "chest_splitter", None)
0209:     if not settings:
0210:         return None, "NOT_FOUND"
0211: 
0212:     # 1. Pelos objetos selecionados (detecta ambiguidade de múltiplas operações - T41 / V21)
0213:     selected_ops = set()
0214:     for o in context.selected_objects:
0215:         op_id = o.get("chest_operation_id")
0216:         if op_id:
0217:             selected_ops.add(op_id)
0218: 
0219:     # Se há mais de uma operação selecionada simultaneamente, é ambíguo!
0220:     if len(selected_ops) > 1:
0221:         return None, "AMBIGUOUS"
0222: 
0223:     if len(selected_ops) == 1:
0224:         single_op_id = next(iter(selected_ops))
0225:         rec = find_operation_record(scene, single_op_id)
0226:         if rec:
0227:             return rec, "FOUND"
0228: 
0229:     # 2. Pelo objeto ativo
0230:     act_obj = context.active_object
0231:     if act_obj:
0232:         op_id = act_obj.get("chest_operation_id")
0233:         if op_id:
0234:             rec = find_operation_record(scene, op_id)
0235:             if rec:
0236:                 return rec, "FOUND"
0237: 
0238:     # 3. Pelo ID ativo na sessão
0239:     active_op_id = getattr(settings, "active_operation_id", "") or getattr(settings, "session_id", "")
0240:     if active_op_id:
0241:         rec = find_operation_record(scene, active_op_id)
0242:         if rec:
0243:             return rec, "FOUND"
0244: 
0245:     return None, "NOT_FOUND"

0248: def resolve_operation(context: bpy.types.Context) -> Optional[Any]:
0249:     """Identifica de forma determinística e inequívoca a operação em foco (B14 / T41)."""
0250:     rec, status = resolve_operation_status(context)
0251:     return rec if status == "FOUND" else None

0254: def check_child_operations(scene: bpy.types.Scene, op_record: Any) -> Tuple[bool, str]:
0255:     """Verifica se existem operações filhas ativas (confirmadas ou prévias) que dependem desta operação (T44)."""
0256:     if not op_record or not hasattr(scene, "chest_operations"):
0257:         return False, ""
0258: 
0259:     parent_id = op_record.operation_id
0260:     loc = _get_current_locale()
0261:     for other in scene.chest_operations:
0262:         if other.operation_id != parent_id and other.parent_operation_id == parent_id:
0263:             if other.status == 'COMMITTED':
0264:                 msg = (
0265:                     f"A operação possui cortes posteriores confirmados ativos (ID: {other.operation_id[:8]}). "
0266:                     f"É necessário reverter primeiro o corte posterior antes de reverter o corte pai."
0267:                     if loc == "pt_BR"
0268:                     else f"The operation has active committed child cuts (ID: {other.operation_id[:8]}). "
0269:                          f"You must revert the posterior cut before reverting this parent cut."
0270:                 )
0271:                 return True, msg
0272:             elif other.status == 'PREVIEW':
0273:                 msg = (
0274:                     f"A operação possui uma prévia de corte filha ativa (ID: {other.operation_id[:8]}). "
0275:                     f"Remova ou descarte a prévia filha antes de reverter o corte pai."
0276:                     if loc == "pt_BR"
0277:                     else f"The operation has an active preview child cut (ID: {other.operation_id[:8]}). "
0278:                          f"Remove or discard the child preview before reverting this parent cut."
0279:                 )
0280:                 return True, msg
0281:     return False, ""

0299:             pass

0350:                     objs_to_remove.add(o)
0351: 
0352:     if op_record:

0354:             if ref.obj and ref.obj.name in bpy.data.objects and ref.role in (
0355:                 ROLE_PREVIEW_A, ROLE_PREVIEW_B, ROLE_PREVIEW_PART, ROLE_SUBPART_MATERIAL

0371:     target = op_record.target_object if (op_record and op_record.target_object) else settings.target_object

0441:         )

0445:         op_record = find_operation_record(scene, settings.session_id)

0497:                 bpy.data.meshes.remove(old_mesh, do_unlink=True)

0505:         target.hide_render = op_record.target_hide_render

0509:         rec_name = op_record.target_name or "Restored_Target"
0510:         recovered_obj = backup_snap.copy()
0511:         recovered_obj.data = backup_snap.data.copy()
