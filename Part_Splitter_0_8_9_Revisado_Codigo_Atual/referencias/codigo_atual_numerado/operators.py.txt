0099: def cleanup_collection_objects(

0102:     session_id: Optional[str] = None,
0103: ):
0104:     """Remove de forma segura objetos temporários de prévia sem remover o alvo, cópias de trabalho ou objetos do usuário."""
0105:     if not collection:
0106:         return
0107: 
0108:     exclude_set = set(exclude_objects) if exclude_objects else set()
0109:     exclude_names = {o.name if hasattr(o, "name") else str(o) for o in exclude_set}
0110: 
0111:     scene = bpy.context.scene if bpy.context else None
0112:     target_obj = getattr(getattr(scene, "chest_splitter", None), "target_object", None) if scene else None
0113:     if target_obj:
0114:         exclude_names.add(target_obj.name)
0115: 
0116:     objs_to_process = list(collection.objects)
0117:     for obj in objs_to_process:
0118:         if obj.name in exclude_names:
0119:             # Caso seja o alvo em uma coleção antiga (ex.: 0.8.4), migra com segurança para a coleção de trabalho
0120:             if target_obj and obj.name == target_obj.name and collection.name == COLLECTION_PREVIEWS_NAME:
0121:                 collection.objects.unlink(obj)
0122:                 if scene:
0123:                     work_col = get_or_create_collection(scene, COLLECTION_WORK_NAME)
0124:                     if obj.name not in work_col.objects:
0125:                         work_col.objects.link(obj)
0126:                 obj["chest_role"] = ROLE_WORK_COPY
0127:             continue
0128: 
0129:         # Nunca apaga cópias de trabalho ou objetos marcados com papel protegido
0130:         role = obj.get("chest_role")
0131:         if role == ROLE_WORK_COPY:
0132:             if collection.name == COLLECTION_PREVIEWS_NAME:
0133:                 collection.objects.unlink(obj)
0134:                 if scene:
0135:                     work_col = get_or_create_collection(scene, COLLECTION_WORK_NAME)
0136:                     if obj.name not in work_col.objects:
0137:                         work_col.objects.link(obj)
0138:             continue
0139: 
0140:         # Se houver filtro de session_id, preserva objetos de outras sessões
0141:         obj_sid = obj.get("chest_session_id")
0142:         if session_id and obj_sid and obj_sid != session_id:
0143:             continue
0144: 
0145:         # Identifica se é objeto de preview descartável
0146:         # Antigas prévias (0.8.4) não tinham session_id, então excluímos por nome + coleção apenas se tiver certeza.

0150:             is_preview_obj = True
0151:         elif obj_sid and obj_sid == session_id:
0152:             is_preview_obj = True
0153: 
0154:         if is_preview_obj:
0155:             mesh_data = obj.data if obj.type == 'MESH' else None

0162:             if collection.name == COLLECTION_PREVIEWS_NAME:

0168: def validate_printable_mesh(
0169:     bm: bmesh.types.BMesh,
0170:     part_name: str = "Peça",
0171:     unit_scale: float = 1000.0,
0172:     check_manifold: bool = True,
0173:     min_volume_mm3: float = 1e-3,
0174: ) -> Tuple[bool, str, Dict[str, Any]]:
0175:     """Valida se uma malha atende aos critérios mínimos de solidez para impressão 3D."""
0176:     bm.verts.ensure_lookup_table()
0177:     bm.edges.ensure_lookup_table()
0178:     bm.faces.ensure_lookup_table()
0179: 
0180:     verts_count = len(bm.verts)
0181:     faces_count = len(bm.faces)
0182: 
0183:     if verts_count < 4 or faces_count < 4:
0184:         return False, f"{part_name}: contagem insuficiente de elementos ({faces_count} faces, {verts_count} vértices).", {}
0185: 
0186:     for v in bm.verts:
0187:         if not (math.isfinite(v.co.x) and math.isfinite(v.co.y) and math.isfinite(v.co.z)):
0188:             return False, f"{part_name}: contém coordenadas infinitas ou NaN.", {}
0189: 
0190:     vol_bu = abs(bm.calc_volume(signed=True))
0191:     vol_mm3 = vol_bu * (unit_scale ** 3)
0192:     if vol_mm3 < min_volume_mm3:
0193:         return False, f"{part_name}: volume nulo ou desprezível ({vol_mm3:.4f} mm³).", {}
0194: 
0195:     non_manifold_edges = [e for e in bm.edges if not e.is_manifold]
0196:     if check_manifold and len(non_manifold_edges) > 0:
0197:         return False, f"{part_name}: possui {len(non_manifold_edges)} aresta(s) não-manifold/abertas (não é um sólido fechado).", {}
0198: 
0199:     metrics = evaluate_bmesh_metrics(bm, unit_scale=unit_scale)
0200:     return True, "OK", metrics

0204: def get_target_bounds(obj: bpy.types.Object):
0205:     """Calcula o centro e o raio aproximado do bounding box do objeto em coordenadas globais."""
0206:     bbox_corners = [obj.matrix_world @ Vector(corner) for corner in obj.bound_box]
0207:     xs = [c.x for c in bbox_corners]
0208:     ys = [c.y for c in bbox_corners]
0209:     zs = [c.z for c in bbox_corners]
0210: 
0211:     center = Vector(((min(xs) + max(xs)) / 2.0, (min(ys) + max(ys)) / 2.0, (min(zs) + max(zs)) / 2.0))
0212:     size = max(max(xs) - min(xs), max(ys) - min(ys), max(zs) - min(zs))
0213:     if size < 1e-4:
0214:         size = 1.0
0215:     return center, size
