0054: def _on_cutter_object_update(self, context):
0055:     """Atualiza diagnósticos imediatos ao alterar o objeto cortador."""
0056:     if self.solid_cutter_object and self.solid_cutter_object.type == 'MESH':
0057:         try:
0058:             from .diagnostic import get_scene_scale_to_mm
0059:             import bmesh
0060:             scale_to_mm = get_scene_scale_to_mm(context.scene if context else None)
0061:             bm = bmesh.new()
0062:             bm.from_mesh(self.solid_cutter_object.data)
0063:             bm.transform(self.solid_cutter_object.matrix_world)
0064:             self.solid_cutter_is_manifold = all(e.is_manifold for e in bm.edges)
0065:             self.solid_cutter_non_manifold_edges = len([e for e in bm.edges if not e.is_manifold])
0066:             self.solid_cutter_triangles = sum(len(f.verts) - 2 for f in bm.faces)
0067:             vol_bu = bm.calc_volume()
0068:             self.solid_cutter_volume_mm3 = abs(vol_bu) * (scale_to_mm ** 3)
0069:             xs = [v.co.x for v in bm.verts]
0070:             ys = [v.co.y for v in bm.verts]
0071:             zs = [v.co.z for v in bm.verts]
0072:             if xs:
0073:                 self.solid_cutter_dims_mm = (
0074:                     (max(xs) - min(xs)) * scale_to_mm,
0075:                     (max(ys) - min(ys)) * scale_to_mm,
0076:                     (max(zs) - min(zs)) * scale_to_mm,
0077:                 )
0078:             bm.free()
0079:         except Exception:
0080:             pass
0081:     _on_param_change_mark_stale(self, context)
