0025: def point_in_polygon_2d(x: float, y: float, poly: List[Tuple[float, float]]) -> bool:
0026:     """Teste de inclusão de ponto 2D em polígono via Ray Casting (Teorema de Jordan)."""
0027:     inside = False
0028:     n = len(poly)
0029:     if n < 3:
0030:         return False
0031:     j = n - 1
0032:     for i in range(n):
0033:         xi, yi = poly[i]
0034:         xj, yj = poly[j]
0035:         if ((yi > y) != (yj > y)) and (x < (xj - xi) * (y - yi) / (yj - yi + 1e-12) + xi):
0036:             inside = not inside
0037:         j = i
0038:     return inside

0041: def create_capsule_2d_profile(w: float, half_len: float, segments: int = 32, offset: float = 0.0) -> List[Tuple[float, float]]:
0042:     """Gera pontos 2D fechados e ordenados para o perfil de estádio/cápsula com suporte a offset uniforme.
0043:     
0044:     Inclui os pontos finais de tangência para que os lados retos sejam estritamente paralelos,
0045:     garantindo folga perimetral uniforme sem segmentos diagonais indevidos.
0046:     """
0047:     half_segs = max(8, segments // 2)
0048:     r = max(1e-5, w - offset)
0049:     if half_len < 1e-6:
0050:         # Círculo perfeito
0051:         return [(r * math.cos(2.0 * math.pi * i / segments), r * math.sin(2.0 * math.pi * i / segments)) for i in range(segments)]
0052: 
0053:     pts = []
0054:     # Arco superior: centro (0, half_len), ângulo de 0 a pi (incluindo extremos tangentes)
0055:     for i in range(half_segs + 1):
0056:         ang = math.pi * (i / half_segs)
0057:         pts.append((r * math.cos(ang), half_len + r * math.sin(ang)))
0058:     # Arco inferior: centro (0, -half_len), ângulo de pi a 2*pi (incluindo extremos tangentes)
0059:     for i in range(half_segs + 1):
0060:         ang = math.pi + math.pi * (i / half_segs)
0061:         pts.append((r * math.cos(ang), -half_len + r * math.sin(ang)))
0062:     return pts

0085:     bm = bmesh.new()

0087:     eff_radius = (diameter / 2.0) + (clearance_side if is_cavity else 0.0)
0088:     eff_length = length + (clearance_end if is_cavity else 0.0)
0089:     eff_chamfer = 0.0 if is_cavity else min(chamfer, eff_length * 0.4, eff_radius * 0.7)
0090: 
0091:     # Tolerância proporcional de sobreposição na base para garantir operação booleana não coplanar
0092:     base_offset = min(eff_length * 0.05, eff_radius * 0.1)
0093:     z_min = -base_offset
0094:     z_max = eff_length
0095:     z_shoulder = z_max - eff_chamfer
0096: 
0097:     base_verts = []
0098:     shoulder_verts = []
0099:     tip_verts = []
0100: 
0101:     if connector_type == 'CAPSULE':
0102:         # Chave tipo cápsula / estádio
0103:         w = eff_radius
0104:         eff_cap_len = max(diameter, capsule_length) if capsule_length > 0.0 else (diameter * 2.4)
0105:         # Centros dos semicírculos ficam separados por L - D
0106:         half_len = (eff_cap_len - diameter) / 2.0
0107: 
0108:         prof = create_capsule_2d_profile(w, half_len, segments=segments)
0109:         prof_tip = create_capsule_2d_profile(w, half_len, segments=segments, offset=eff_chamfer) if eff_chamfer > 0.0 else []
0110: 
0111:         for idx, (x, y) in enumerate(prof):
0112:             base_verts.append(bm.verts.new((x, y, z_min)))
0113:             shoulder_verts.append(bm.verts.new((x, y, z_shoulder)))
0114:             if eff_chamfer > 0.0:
0115:                 tx, ty = prof_tip[idx]
0116:                 tip_verts.append(bm.verts.new((tx, ty, z_max)))
0117: 
0118:     else:
0119:         # Padrão: Pino Cilíndrico
0120:         tip_r = max(eff_radius * 0.2, eff_radius - eff_chamfer)
0121: 
0122:         for i in range(segments):
0123:             ang = 2.0 * math.pi * i / segments
0124:             cos_a = math.cos(ang)
0125:             sin_a = math.sin(ang)
0126: 
0127:             base_verts.append(bm.verts.new((eff_radius * cos_a, eff_radius * sin_a, z_min)))
0128:             shoulder_verts.append(bm.verts.new((eff_radius * cos_a, eff_radius * sin_a, z_shoulder)))
0129:             if eff_chamfer > 0.0:
0130:                 tip_verts.append(bm.verts.new((tip_r * cos_a, tip_r * sin_a, z_max)))
0131: 
0132:     # Face inferior (aponta para -Z)
0133:     bm.faces.new(reversed(base_verts))
0134: 
0135:     # Paredes laterais
0136:     num_v = len(base_verts)
0137:     for i in range(num_v):
0138:         i_next = (i + 1) % num_v
0139:         bm.faces.new((base_verts[i], base_verts[i_next], shoulder_verts[i_next], shoulder_verts[i]))
0140:         if eff_chamfer > 0.0:
0141:             bm.faces.new((shoulder_verts[i], shoulder_verts[i_next], tip_verts[i_next], tip_verts[i]))
0142: 
0143:     # Topo (fechamento superior)
0144:     bm.faces.new(tip_verts if eff_chamfer > 0.0 else shoulder_verts)
0145: 
0146:     bmesh.ops.recalc_face_normals(bm, faces=bm.faces[:])
0147: 
0148:     mesh_data = bpy.data.meshes.new(name)
0149:     bm.to_mesh(mesh_data)
0150:     bm.free()
0151:     return mesh_data

0182:         v_dir = norm.cross(u_dir).normalized()

0201:                 if hit_loc is None:
0202:                     # Raio sem interseção: não há parede receptora na direção de inserção
0203:                     bm.free()

0205: 
0206:                 avail_depth = (hit_loc - sample_p).length
0207:                 if avail_depth < min_found_depth:
0208:                     min_found_depth = avail_depth
0209: 
0210:                 if avail_depth < (required_depth_bu + min_wall_bu):
0211:                     bm.free()

0213: 
0214:         bm.free()

0217:         eval_obj.to_mesh_clear()

0445:                         )

0484: def validate_connectors(
0485:     positions: List[Vector],
0486:     diameter_mm: float,
0487:     clearance_per_side_mm: float,
0488:     unit_scale: float = 1000.0,
0489:     connector_type: str = 'CYLINDER',
0490:     capsule_length_mm: float = 0.0,
0491:     obj_female: Optional[bpy.types.Object] = None,
0492:     normal: Optional[Vector] = None,
0493:     length_mm: float = 0.0,
0494:     end_clearance_mm: float = 0.0,
0495: ):
0496:     """Valida distâncias entre encaixes e espessura de parede remanescente na peça receptora."""
0497:     eff_length = capsule_length_mm if (connector_type == 'CAPSULE' and capsule_length_mm > diameter_mm) else diameter_mm
0498:     min_dist_bu = (eff_length + 2.0 * clearance_per_side_mm) / unit_scale
0499: 
0500:     for i in range(len(positions)):
0501:         for j in range(i + 1, len(positions)):
0502:             dist = (positions[i] - positions[j]).length
0503:             if dist < min_dist_bu:
0504:                 dist_mm = dist * unit_scale
0505:                 req_mm = eff_length + 2.0 * clearance_per_side_mm
0506:                 raise ConnectorValidationError(
0507:                     f"Sobreposição entre encaixes detectada: distância {dist_mm:.2f} mm < mínimo {req_mm:.2f} mm."
0508:                 )
0509: 
0510:     # Validação de espessura de parede da peça receptora (Critério C5)
0511:     if obj_female is not None and normal is not None and length_mm > 0.0:
0512:         req_depth_bu = (length_mm + end_clearance_mm) / unit_scale
0513:         cavity_r_bu = ((diameter_mm / 2.0) + clearance_per_side_mm) / unit_scale
0514:         ok, avail = check_receiving_wall_thickness(
0515:             obj_female,
0516:             positions,
0517:             normal,
0518:             req_depth_bu,
0519:             cavity_radius_bu=cavity_r_bu,
0520:         )
0521:         if not ok:
0522:             avail_mm = avail * unit_scale
0523:             tot_depth_mm = length_mm + end_clearance_mm
0524:             if avail <= 1e-6:
0525:                 raise ConnectorValidationError(
0526:                     "Parede receptora inválida ou ausente: a cavidade não encontra corpo sólido suficiente "
0527:                     "na direção de inserção. Verifique a orientação ou integridade da peça receptora."
0528:                 )
0529:             else:
0530:                 raise ConnectorValidationError(
0531:                     f"Parede fina: a cavidade de {tot_depth_mm:.1f} mm atravessaria a peça receptora "
0532:                     f"(espessura disponível: {avail_mm:.1f} mm). Reduza o comprimento de inserção."
0533:                 )

0537: def apply_connectors_boolean(
0538:     obj_male: bpy.types.Object,
0539:     obj_female: bpy.types.Object,
0540:     plane_normal: Vector,
0541:     positions: List[Vector],
0542:     connector_type: str,
0543:     diameter_mm: float,
0544:     length_mm: float,
0545:     chamfer_mm: float,
0546:     clearance_side_mm: float,
0547:     end_clearance_mm: float,
0548:     unit_scale: float = 1000.0,
0549:     capsule_length_mm: float = 0.0,
0550:     major_axis: Optional[Vector] = None,
0551: ):
0552:     """Executa as operações booleanas para unir os machos e subtrair as cavidades de forma transacional."""
0553:     norm = plane_normal.normalized()
0554: 
0555:     # Validação prévia de espessura de parede da peça receptora
0556:     validate_connectors(
0557:         positions=positions,
0558:         diameter_mm=diameter_mm,
0559:         clearance_per_side_mm=clearance_side_mm,
0560:         unit_scale=unit_scale,
0561:         connector_type=connector_type,
0562:         capsule_length_mm=capsule_length_mm,
0563:         obj_female=obj_female,
0564:         normal=norm,
0565:         length_mm=length_mm,
0566:         end_clearance_mm=end_clearance_mm,
0567:     )
0568: 
0569:     # Determina o eixo maior (direção de alinhamento das cápsulas no plano de corte)
0570:     if major_axis is None:
0571:         if len(positions) >= 2:
0572:             delta = positions[1] - positions[0]
0573:             delta_proj = delta - norm * delta.dot(norm)
0574:             if delta_proj.length > 1e-6:
0575:                 major_axis = delta_proj.normalized()
0576:         if major_axis is None:
0577:             ref_up = Vector((0, 0, 1)) if abs(norm.z) < 0.9 else Vector((0, 1, 0))
0578:             major_axis = norm.cross(ref_up).normalized()
0579: 
0580:     # Constrói sistema ortonormal estável: local_z = norm, local_y = major_axis, local_x = local_y x local_z
0581:     local_z = norm.normalized()
0582:     local_y = (major_axis - local_z * major_axis.dot(local_z)).normalized()
0583:     local_x = local_y.cross(local_z).normalized()
0584:     rot_mat = Matrix((local_x, local_y, local_z)).transposed()
0585:     pin_rot = rot_mat.to_euler()
0586: 
0587:     cutters_to_clean = []
0588:     modifiers_male_added = []
0589:     modifiers_female_added = []
0590: 
0591:     temp_col = obj_male.users_collection[0] if obj_male.users_collection else bpy.context.scene.collection
0592: 
0593:     diam_bu = diameter_mm / unit_scale
0594:     len_bu = length_mm / unit_scale
0595:     chamf_bu = chamfer_mm / unit_scale
0596:     clear_side_bu = clearance_side_mm / unit_scale
0597:     clear_end_bu = end_clearance_mm / unit_scale
0598:     cap_len_bu = (capsule_length_mm / unit_scale) if capsule_length_mm > 0 else 0.0
0599: 
0600:     try:
0601:         # 1. Cria pinos para obj_male
0602:         for idx, pos in enumerate(positions):
0603:             pin_mesh = create_pin_mesh_data(
0604:                 name=f"temp_pin_{idx}",
0605:                 connector_type=connector_type,
0606:                 diameter=diam_bu,
0607:                 length=len_bu,
0608:                 chamfer=chamf_bu,
0609:                 is_cavity=False,
0610:                 capsule_length=cap_len_bu,
0611:             )
0612:             pin_obj = bpy.data.objects.new(f"temp_pin_{idx}", pin_mesh)
0613:             for mat in obj_male.data.materials:
0614:                 pin_mesh.materials.append(mat)
0615:             pin_obj.location = pos
0616:             pin_obj.rotation_euler = pin_rot
0617:             temp_col.objects.link(pin_obj)
0618:             cutters_to_clean.append(pin_obj)
0619: 
0620:             mod_name = f"Pin_Union_{idx}"
0621:             mod_union = obj_male.modifiers.new(name=mod_name, type='BOOLEAN')
0622:             mod_union.operation = 'UNION'
0623:             mod_union.solver = 'EXACT'
0624:             mod_union.object = pin_obj
0625:             modifiers_male_added.append(mod_name)
0626: 
0627:         # 2. Cria cavidades para obj_female
0628:         for idx, pos in enumerate(positions):
0629:             cavity_mesh = create_pin_mesh_data(
0630:                 name=f"temp_cavity_{idx}",
0631:                 connector_type=connector_type,
0632:                 diameter=diam_bu,
0633:                 length=len_bu,
0634:                 chamfer=0.0,
0635:                 is_cavity=True,
0636:                 clearance_side=clear_side_bu,
0637:                 clearance_end=clear_end_bu,
0638:                 capsule_length=cap_len_bu,
0639:             )
0640:             cavity_obj = bpy.data.objects.new(f"temp_cavity_{idx}", cavity_mesh)
0641:             for mat in obj_female.data.materials:
0642:                 cavity_mesh.materials.append(mat)
0643:             cavity_obj.location = pos
0644:             cavity_obj.rotation_euler = pin_rot
0645:             temp_col.objects.link(cavity_obj)
0646:             cutters_to_clean.append(cavity_obj)
0647: 
0648:             mod_name = f"Cavity_Diff_{idx}"
0649:             mod_diff = obj_female.modifiers.new(name=mod_name, type='BOOLEAN')
0650:             mod_diff.operation = 'DIFFERENCE'
0651:             mod_diff.solver = 'EXACT'
0652:             mod_diff.object = cavity_obj
0653:             modifiers_female_added.append(mod_name)
0654: 
0655:         # Atualiza a view layer para garantir avaliação correta
0656:         bpy.context.view_layer.update()
0657:         depsgraph = bpy.context.evaluated_depsgraph_get()
0658: 
0659:         eval_male = obj_male.evaluated_get(depsgraph)
0660:         new_male_mesh = eval_male.to_mesh()
0661: 
0662:         eval_female = obj_female.evaluated_get(depsgraph)
0663:         new_female_mesh = eval_female.to_mesh()
0664: 
0665:         try:
0666:             # Validação de integridade das malhas avaliadas
0667:             if len(new_male_mesh.polygons) == 0 or len(new_female_mesh.polygons) == 0:
0668:                 raise RuntimeError("Operação booleana de encaixe gerou malha vazia ou corrompida.")
0669: 
0670:             # Transfere malhas avaliadas em definitivo
0671:             old_mesh_male = obj_male.data
0672:             final_male = new_male_mesh.copy()
0673:             obj_male.data = final_male
0674: 
0675:             old_mesh_female = obj_female.data
0676:             final_female = new_female_mesh.copy()
0677:             obj_female.data = final_female
0678:         finally:
0679:             eval_male.to_mesh_clear()
0680:             eval_female.to_mesh_clear()
0681: 
0682:         # Remove slots de material vazios
0683:         for obj in (obj_male, obj_female):
0684:             empty_slots = [i for i, m in enumerate(obj.data.materials) if m is None]
0685:             for i in reversed(empty_slots):
0686:                 obj.data.materials.pop(index=i)
0687: 
0688:         if old_mesh_male and old_mesh_male.users == 0:
0689:             bpy.data.meshes.remove(old_mesh_male, do_unlink=True)
0690:         if old_mesh_female and old_mesh_female.users == 0:
0691:             bpy.data.meshes.remove(old_mesh_female, do_unlink=True)
0692: 
0693:     finally:
0694:         # Garante remoção dos modificadores adicionados (sem tocar nos modificadores do usuário)
0695:         for mod_name in modifiers_male_added:
0696:             mod = obj_male.modifiers.get(mod_name)
0697:             if mod:
0698:                 obj_male.modifiers.remove(mod)
0699: 
0700:         for mod_name in modifiers_female_added:
0701:             mod = obj_female.modifiers.get(mod_name)
0702:             if mod:
0703:                 obj_female.modifiers.remove(mod)
0704: 
0705:         # Limpa objetos e malhas auxiliares de corte
0706:         for cutter in cutters_to_clean:
0707:             c_mesh = cutter.data
0708:             if cutter.name in temp_col.objects:
0709:                 temp_col.objects.unlink(cutter)
0710:             if cutter.name in bpy.data.objects:
0711:                 bpy.data.objects.remove(cutter, do_unlink=True)
0712:             if c_mesh and c_mesh.users == 0:
0713:                 bpy.data.meshes.remove(c_mesh, do_unlink=True)
