0467: def analyze_loop_region(
0468:     obj: bpy.types.Object,
0469:     indices: List[int],
0470:     seed_face_idx: int,
0471:     scene: Optional[bpy.types.Scene] = None,
0472: ) -> Dict[str, Any]:
0473:     """Analisa a região delimitada pelo loop para decisão automática entre DETACH e EXTRUDE."""
0474:     scale_to_mm = get_scene_scale_to_mm(scene) if scene else 1000.0
0475:     matrix_world = obj.matrix_world.copy()
0476: 
0477:     bm = bmesh.new()
0478:     bm.from_mesh(obj.data)
0479:     bm.verts.ensure_lookup_table()
0480:     bm.edges.ensure_lookup_table()
0481:     bm.faces.ensure_lookup_table()
0482: 
0483:     if seed_face_idx < 0 or seed_face_idx >= len(bm.faces):
0484:         bm.free()
0485:         raise ValueError("Face-semente inválida ou não selecionada.")
0486: 
0487:     # Mapeia as arestas do contorno
0488:     loop_verts_local = [bm.verts[idx] for idx in indices]
0489:     loop_edges = set()
0490:     n_loop = len(loop_verts_local)
0491:     for i in range(n_loop):
0492:         v1 = loop_verts_local[i]
0493:         v2 = loop_verts_local[(i + 1) % n_loop]
0494:         edge = bm.edges.get((v1, v2))
0495:         if edge:
0496:             loop_edges.add(edge)
0497: 
0498:     # Flood fill para coletar faces de A
0499:     seed_face = bm.faces[seed_face_idx]
0500:     faces_a = set()
0501:     queue = [seed_face]
0502:     faces_a.add(seed_face)
0503: 
0504:     while queue:
0505:         curr_face = queue.pop(0)
0506:         for edge in curr_face.edges:
0507:             if edge in loop_edges:
0508:                 continue
0509:             for adj_face in edge.link_faces:
0510:                 if adj_face not in faces_a:
0511:                     faces_a.add(adj_face)
0512:                     queue.append(adj_face)
0513: 
0514:     loop_coords_world = [matrix_world @ v.co for v in loop_verts_local]
0515:     centroid = sum(loop_coords_world, Vector()) / len(loop_coords_world)
0516: 
0517:     # Normal média do loop via Newell
0518:     n = Vector((0.0, 0.0, 0.0))
0519:     n_pts = len(loop_coords_world)
0520:     for i in range(n_pts):
0521:         v1 = loop_coords_world[i]
0522:         v2 = loop_coords_world[(i + 1) % n_pts]
0523:         n.x += (v1.y - v2.y) * (v1.z + v2.z)
0524:         n.y += (v1.z - v2.z) * (v1.x + v2.x)
0525:         n.z += (v1.x - v2.x) * (v1.y + v2.y)
0526:     if n.length > 1e-6:
0527:         n.normalize()
0528:     else:
0529:         n = Vector((0.0, 0.0, 1.0))
0530: 
0531:     # Determina a direção para dentro do sólido
0532:     normal_matrix = matrix_world.to_3x3().inverted().transposed()
0533:     seed_norm_world = (normal_matrix @ seed_face.normal).normalized()
0534: 
0535:     # Prepara BVHTree no espaço de mundo para testar direção e espessura da parede
0536:     bm_bvh = bm.copy()
0537:     bm_bvh.transform(matrix_world)
0538:     bvh = BVHTree.FromBMesh(bm_bvh)
0539: 
0540:     # Inward candidato primário: o oposto da normal externa da face semente
0541:     cand_inward = -seed_norm_world
0542:     hit_c1, _, _, _ = bvh.ray_cast(centroid + cand_inward * 1e-4, cand_inward)
0543:     hit_c2, _, _, _ = bvh.ray_cast(centroid - cand_inward * 1e-4, -cand_inward)
0544: 
0545:     if hit_c1 is not None and hit_c2 is None:
0546:         inward_dir = cand_inward
0547:     elif hit_c2 is not None and hit_c1 is None:
0548:         inward_dir = -cand_inward
0549:     elif hit_c1 is not None and hit_c2 is not None:
0550:         # Ambos intersectam: escolhe o que penetra no sólido a partir da face semente
0551:         inward_dir = cand_inward
0552:     else:
0553:         inward_dir = cand_inward
0554: 
0555:     # Analisa relevo (altura) dos vértices da região A em relação ao plano do loop
0556:     verts_a = set(v for f in faces_a for v in f.verts)
0557:     coords_a_world = [matrix_world @ v.co for v in verts_a]
0558:     outward_dir = -inward_dir
0559:     heights = [(p - centroid).dot(outward_dir) for p in coords_a_world]
0560:     max_h = max(heights) if heights else 0.0
0561:     min_h = min(heights) if heights else 0.0
0562:     span_h_bu = max_h - min_h
0563:     span_h_mm = span_h_bu * scale_to_mm
0564: 
0565:     # Avalia espessura disponível na direção inward via BVHTree
0566:     min_wall_depth_bu = 1e9
0567:     test_points = [centroid] + loop_coords_world
0568:     for pt in test_points:
0569:         ray_start = pt + inward_dir * 1e-4
0570:         hit_loc, hit_norm, hit_idx, hit_dist = bvh.ray_cast(ray_start, inward_dir)
0571:         if hit_loc is not None:
0572:             dist = (hit_loc - pt).length
0573:             if dist < min_wall_depth_bu:
0574:                 min_wall_depth_bu = dist
0575: 
0576:     bm_bvh.free()
0577:     bm.free()
0578: 
0579:     max_avail_depth_mm = (min_wall_depth_bu * scale_to_mm) if min_wall_depth_bu < 1e8 else 1000.0
0580: 
0581:     if span_h_mm < 0.2:
0582:         detected_mode = 'EXTRUDE'
0583:         reason = (
0584:             f"Região plana detectada (desvio de relevo: {span_h_mm:.2f} mm). "
0585:             f"Requer profundidade de extração para gerar peça sólida."
0586:         )
0587:         is_planar_region = True
0588:     else:
0589:         detected_mode = 'DETACH'
0590:         reason = (
0591:             f"Saliência 3D detectada (altura saliente: {span_h_mm:.2f} mm). "
0592:             f"Destacando volume pela interface do contorno."
0593:         )
0594:         is_planar_region = False
0595: 
0596:     return {
0597:         "detected_mode": detected_mode,
0598:         "reason": reason,
0599:         "protrusion_height_mm": max(0.0, max_h * scale_to_mm),
0600:         "is_planar_region": is_planar_region,
0601:         "inward_dir_world": inward_dir,
0602:         "loop_normal_world": n,
0603:         "centroid_world": centroid,
0604:         "max_avail_depth_mm": max_avail_depth_mm,
0605:         "coords_world": loop_coords_world,
0606:     }

0609: def create_loop_prism_cutter(
0610:     coords_world: List[Vector],
0611:     inward_dir: Vector,
0612:     depth_bu: float,
0613:     outward_ext_bu: float = 1e-3,
0614:     name: str = "temp_loop_prism",
0615: ) -> bpy.types.Object:
0616:     """Constrói um prisma 3D fechado e manifold a partir de um contorno de coordenadas mundiais."""
0617:     bm = bmesh.new()
0618:     in_norm = inward_dir.normalized()
0619: 
0620:     # Remove pontos colineares (ex: vértice no meio de uma aresta reta do contorno).
0621:     # Eles geram triângulos degenerados nas tampas e quebram o booleano EXACT.
0622:     pts = list(coords_world)
0623:     changed = True
0624:     while changed and len(pts) > 3:
0625:         changed = False
0626:         for i in range(len(pts)):
0627:             p_prev, p, p_next = pts[i - 1], pts[i], pts[(i + 1) % len(pts)]
0628:             d1 = p - p_prev
0629:             d2 = p_next - p
0630:             if d1.length < 1e-9 or d2.length < 1e-9 or d1.normalized().cross(d2.normalized()).length < 1e-5:
0631:                 pts.pop(i)
0632:                 changed = True
0633:                 break
0634:     coords_world = pts
0635:     n = len(coords_world)
0636: 
0637:     top_verts = [bm.verts.new(p - outward_ext_bu * in_norm) for p in coords_world]
0638:     bot_verts = [bm.verts.new(p + depth_bu * in_norm) for p in coords_world]
0639:     bm.verts.ensure_lookup_table()
0640: 
0641:     for i in range(n):
0642:         i_next = (i + 1) % n
0643:         bm.faces.new((top_verts[i], top_verts[i_next], bot_verts[i_next], bot_verts[i]))
0644: 
0645:     f_top = bm.faces.new(reversed(top_verts))
0646:     f_bot = bm.faces.new(bot_verts)
0647: 
0648:     bmesh.ops.triangulate(bm, faces=[f_top, f_bot])
0649:     bmesh.ops.recalc_face_normals(bm, faces=bm.faces[:])
0650: 
0651:     mesh = bpy.data.meshes.new(f"{name}_Mesh")
0652:     bm.to_mesh(mesh)
0653:     bm.free()
0654: 
0655:     obj = bpy.data.objects.new(name, mesh)
0656:     obj.hide_viewport = False
0657:     obj.hide_render = False
0658:     bpy.context.scene.collection.objects.link(obj)
0659:     return obj

0662: def slice_mesh_by_loop_extrude(
0663:     obj: bpy.types.Object,
0664:     target_name: str,

0667:     depth_mm: float,
0668:     through_cut: bool = False,
0669:     scene: Optional[bpy.types.Scene] = None,

0671:     """Corta o modelo extraindo o volume delimitado pelo contorno até a profundidade especificada.

0679:     analysis = analyze_loop_region(obj, indices, seed_face_idx, scene=actual_scene)

0682:     # Validação de espessura (Critério L5)
0683:     if depth_mm > (max_avail_mm + 1e-3) and not through_cut:
0684:         raise ValueError(
0685:             f"A profundidade de extração ({depth_mm:.1f} mm) ultrapassa a espessura da peça "
0686:             f"({max_avail_mm:.1f} mm). Reduza a profundidade ou ative a opção 'Corte Passante'."

0700:         coords_world=coords_world,

0707:     try:
0708:         bpy.context.view_layer.update()
0709:

0717:         mod_a.solver = 'EXACT'
0718:         mod_a.object = prism_obj

0722:         temp_obj_b.hide_viewport = False

0726:         mod_b.operation = 'DIFFERENCE'

0728:         mod_b.object = prism_obj

0732: 
0733:         eval_a = temp_obj_a.evaluated_get(dg).to_mesh()
0734:         eval_b = temp_obj_b.evaluated_get(dg).to_mesh()
0735: 
0736:         bm_a = bmesh.new()
0737:         bm_a.from_mesh(eval_a)
0738:         bm_a.transform(temp_obj_a.matrix_world)
0739: 
0740:         bm_b = bmesh.new()
0741:         bm_b.from_mesh(eval_b)
0742:         bm_b.transform(temp_obj_b.matrix_world)
0743:

0745:         temp_obj_b.to_mesh_clear()

0749:         actual_scene.collection.objects.unlink(temp_obj_a)
0750:         actual_scene.collection.objects.unlink(temp_obj_b)
0751:         bpy.data.objects.remove(temp_obj_a, do_unlink=True)
0752:         bpy.data.objects.remove(temp_obj_b, do_unlink=True)
