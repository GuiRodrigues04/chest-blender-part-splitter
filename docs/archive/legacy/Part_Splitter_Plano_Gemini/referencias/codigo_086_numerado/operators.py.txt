0084: def cleanup_collection_objects(
0085:     collection: bpy.types.Collection,
0086:     exclude_objects: Optional[set] = None,
0087:     session_id: Optional[str] = None,
0088: ):
0089:     """Remove de forma segura objetos temporários de prévia sem remover o alvo, cópias de trabalho ou objetos do usuário."""
0090:     if not collection:
0091:         return
0092: 
0093:     exclude_set = set(exclude_objects) if exclude_objects else set()
0094:     exclude_names = {o.name if hasattr(o, "name") else str(o) for o in exclude_set}
0095: 
0096:     scene = bpy.context.scene if bpy.context else None
0097:     target_obj = getattr(getattr(scene, "chest_splitter", None), "target_object", None) if scene else None
0098:     if target_obj:
0099:         exclude_names.add(target_obj.name)
0100: 
0101:     objs_to_process = list(collection.objects)
0102:     for obj in objs_to_process:
0103:         if obj.name in exclude_names:
0104:             # Caso seja o alvo em uma coleção antiga (ex.: 0.8.4), migra com segurança para a coleção de trabalho
0105:             if target_obj and obj.name == target_obj.name and collection.name == COLLECTION_PREVIEWS_NAME:
0106:                 collection.objects.unlink(obj)
0107:                 if scene:
0108:                     work_col = get_or_create_collection(scene, COLLECTION_WORK_NAME)
0109:                     if obj.name not in work_col.objects:
0110:                         work_col.objects.link(obj)
0111:                 obj["chest_role"] = ROLE_WORK_COPY
0112:             continue
0113: 
0114:         # Nunca apaga cópias de trabalho ou objetos marcados com papel protegido
0115:         role = obj.get("chest_role")
0116:         if role == ROLE_WORK_COPY:
0117:             if collection.name == COLLECTION_PREVIEWS_NAME:
0118:                 collection.objects.unlink(obj)
0119:                 if scene:
0120:                     work_col = get_or_create_collection(scene, COLLECTION_WORK_NAME)
0121:                     if obj.name not in work_col.objects:
0122:                         work_col.objects.link(obj)
0123:             continue
0124: 
0125:         # Se houver filtro de session_id, preserva objetos de outras sessões
0126:         obj_sid = obj.get("chest_session_id")
0127:         if session_id and obj_sid and obj_sid != session_id:
0128:             continue
0129: 
0130:         # Identifica se é objeto de preview descartável
0131:         # Antigas prévias (0.8.4) não tinham session_id, então excluímos por nome + coleção apenas se tiver certeza.
0132:         # Mas para proteger objetos do usuário, exigimos que tenha o "chest_role" OU "chest_session_id" se o nome bater.
0133:         is_preview_obj = False
0134:         if role in (ROLE_PREVIEW_A, ROLE_PREVIEW_B, ROLE_PREVIEW_PART, ROLE_SUBPART_MATERIAL):
0135:             is_preview_obj = True
0136:         elif obj_sid and obj_sid == session_id:
0137:             is_preview_obj = True
0138: 
0139:         if is_preview_obj:
0140:             mesh_data = obj.data if obj.type == 'MESH' else None
0141:             collection.objects.unlink(obj)
0142:             bpy.data.objects.remove(obj, do_unlink=True)
0143:             if mesh_data and mesh_data.users == 0:
0144:                 bpy.data.meshes.remove(mesh_data, do_unlink=True)
0145:         else:
0146:             # Não é prévia. Se está na coleção temporária, move para a coleção principal da cena para não sumir
0147:             if collection.name == COLLECTION_PREVIEWS_NAME:
0148:                 if scene and obj.name not in scene.collection.objects:
0149:                     scene.collection.objects.link(obj)

0153: def validate_printable_mesh(
0154:     bm: bmesh.types.BMesh,
0155:     part_name: str = "Peça",
0156:     unit_scale: float = 1000.0,
0157:     check_manifold: bool = True,
0158:     min_volume_mm3: float = 1e-3,
0159: ) -> Tuple[bool, str, Dict[str, Any]]:
0160:     """Valida se uma malha atende aos critérios mínimos de solidez para impressão 3D."""
0161:     bm.verts.ensure_lookup_table()
0162:     bm.edges.ensure_lookup_table()
0163:     bm.faces.ensure_lookup_table()
0164: 
0165:     verts_count = len(bm.verts)
0166:     faces_count = len(bm.faces)
0167: 
0168:     if verts_count < 4 or faces_count < 4:
0169:         return False, f"{part_name}: contagem insuficiente de elementos ({faces_count} faces, {verts_count} vértices).", {}
0170: 
0171:     for v in bm.verts:
0172:         if not (math.isfinite(v.co.x) and math.isfinite(v.co.y) and math.isfinite(v.co.z)):
0173:             return False, f"{part_name}: contém coordenadas infinitas ou NaN.", {}
0174: 
0175:     vol_bu = abs(bm.calc_volume(signed=True))
0176:     vol_mm3 = vol_bu * (unit_scale ** 3)
0177:     if vol_mm3 < min_volume_mm3:
0178:         return False, f"{part_name}: volume nulo ou desprezível ({vol_mm3:.4f} mm³).", {}
0179: 
0180:     non_manifold_edges = [e for e in bm.edges if not e.is_manifold]
0181:     if check_manifold and len(non_manifold_edges) > 0:
0182:         return False, f"{part_name}: possui {len(non_manifold_edges)} aresta(s) não-manifold/abertas (não é um sólido fechado).", {}
0183: 
0184:     metrics = evaluate_bmesh_metrics(bm, unit_scale=unit_scale)
0185:     return True, "OK", metrics

0189: def get_target_bounds(obj: bpy.types.Object):
0190:     """Calcula o centro e o raio aproximado do bounding box do objeto em coordenadas globais."""
0191:     bbox_corners = [obj.matrix_world @ Vector(corner) for corner in obj.bound_box]
0192:     xs = [c.x for c in bbox_corners]
0193:     ys = [c.y for c in bbox_corners]
0194:     zs = [c.z for c in bbox_corners]
0195: 
0196:     center = Vector(((min(xs) + max(xs)) / 2.0, (min(ys) + max(ys)) / 2.0, (min(zs) + max(zs)) / 2.0))
0197:     size = max(max(xs) - min(xs), max(ys) - min(ys), max(zs) - min(zs))
0198:     if size < 1e-4:
0199:         size = 1.0
0200:     return center, size
