0033: def _on_param_change_mark_stale(self, context):
0034:     """Callback para marcar o preview como desatualizado quando parâmetros de corte ou encaixe mudam."""

0036:         self.session_status = 'PREVIEW_STALE'

0053: def _on_cutter_object_update(self, context):
0054:     """Atualiza diagnósticos imediatos ao alterar o objeto cortador."""
0055:     if self.solid_cutter_object and self.solid_cutter_object.type == 'MESH':
0056:         try:
0057:             from .diagnostic import get_scene_scale_to_mm
0058:             import bmesh
0059:             scale_to_mm = get_scene_scale_to_mm(context.scene if context else None)
0060:             bm = bmesh.new()
0061:             bm.from_mesh(self.solid_cutter_object.data)
0062:             bm.transform(self.solid_cutter_object.matrix_world)
0063:             self.solid_cutter_is_manifold = all(e.is_manifold for e in bm.edges)
0064:             self.solid_cutter_non_manifold_edges = len([e for e in bm.edges if not e.is_manifold])
0065:             self.solid_cutter_triangles = sum(len(f.verts) - 2 for f in bm.faces)
0066:             vol_bu = bm.calc_volume()
0067:             self.solid_cutter_volume_mm3 = abs(vol_bu) * (scale_to_mm ** 3)
0068:             xs = [v.co.x for v in bm.verts]
0069:             ys = [v.co.y for v in bm.verts]
0070:             zs = [v.co.z for v in bm.verts]
0071:             if xs:
0072:                 self.solid_cutter_dims_mm = (
0073:                     (max(xs) - min(xs)) * scale_to_mm,
0074:                     (max(ys) - min(ys)) * scale_to_mm,
0075:                     (max(zs) - min(zs)) * scale_to_mm,
0076:                 )
0077:             bm.free()
0078:         except Exception:
0079:             pass
0080:     _on_param_change_mark_stale(self, context)

0085:     _on_param_change_mark_stale(self, context)
