0288: def _get_current_locale(forced_locale: Optional[str] = None) -> str:
0289:     """Retorna o código de idioma atual do Blender (pt_BR ou fallback en_US)."""
0290:     try:
0291:         loc = forced_locale if forced_locale is not None else getattr(bpy.app.translations, "locale", None)
0292:         if loc and loc.startswith("pt"):
0293:             return "pt_BR"
0294:     except Exception:
0295:         pass
0296:     return "en_US"  # Padrão internacional em inglês

0299: def get_help_tooltip(topic: str) -> str:
0300:     """Retorna o texto de tooltip resumido para o tópico indicado."""
0301:     loc = _get_current_locale()
0302:     topic_data = HELP_TOPICS.get(topic)
0303:     if not topic_data:
0304:         return "Clique para ajuda sobre este recurso."
0305:     tooltips = topic_data.get("tooltip", {})
0306:     return tooltips.get(loc, tooltips.get("pt_BR", ""))
0307: 
0308: 
0309: def get_help_popup_content(topic: str) -> Tuple[str, List[str]]:
0310:     """Retorna título e linhas para exibição em popup dialog."""
0311:     loc = _get_current_locale()
0312:     topic_data = HELP_TOPICS.get(topic)
0313:     if not topic_data:
0314:         return ("Ajuda", ["Nenhum tópico explicativo disponível."])
0315:     
0316:     title = topic_data.get("title", {}).get(loc, topic_data.get("title", {}).get("pt_BR", "Ajuda"))
0317:     lines = topic_data.get("popup_lines", {}).get(loc, topic_data.get("popup_lines", {}).get("pt_BR", []))
0318:     return title, lines
